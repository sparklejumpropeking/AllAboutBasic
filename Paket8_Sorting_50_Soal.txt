Soal 1. Barisan Bibit Pohon (Bubble Sort Naik)
----------------------------------------------------------------------

Petani menanam bibit dari yang paling pendek. Urutkan tinggi bibit secara naik dengan bubble sort.

[CONTOH TAMPILAN]
Jumlah bibit: 6
Tinggi bibit (cm): 34 12 45 7 23 19
Urutan tanam: 7 12 19 23 34 45

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int tinggiBibit[100], jumlahBibit;
    cout << "Jumlah bibit: ";
    cin >> jumlahBibit;
    cout << "Tinggi bibit (cm): ";
    for (int i = 0; i < jumlahBibit; i++) cin >> tinggiBibit[i];
    for (int putaran = 0; putaran < jumlahBibit - 1; putaran++) {
        for (int j = 0; j < jumlahBibit - 1 - putaran; j++) {
            if (tinggiBibit[j] > tinggiBibit[j + 1]) {
                int tampungBibit = tinggiBibit[j];
                tinggiBibit[j] = tinggiBibit[j + 1];
                tinggiBibit[j + 1] = tampungBibit;
            }
        }
    }
    cout << "Urutan tanam:";
    for (int i = 0; i < jumlahBibit; i++) cout << " " << tinggiBibit[i];
    cout << endl;
    return 0;
}

Soal 4. Petugas Arsip yang Malas (Bubble Sort dengan Flag)
----------------------------------------------------------------------

Petugas arsip berhenti mengurutkan begitu satu putaran penuh tidak ada pertukaran. Hitung jumlah putaran dan jumlah pertukaran yang terjadi.

[CONTOH TAMPILAN]
Jumlah arsip: 6
Nomor arsip: 2 3 1 4 5 6
Arsip urut: 1 2 3 4 5 6
Putaran: 3, pertukaran: 2

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int nomorArsip[100], jumlahArsip, totalTukarArsip = 0, putaranDipakai = 0;
    cout << "Jumlah arsip: ";
    cin >> jumlahArsip;
    cout << "Nomor arsip: ";
    for (int i = 0; i < jumlahArsip; i++) cin >> nomorArsip[i];
    for (int putaran = 0; putaran < jumlahArsip - 1; putaran++) {
        bool adaPertukaran = false;
        putaranDipakai++;
        for (int j = 0; j < jumlahArsip - 1 - putaran; j++) {
            if (nomorArsip[j] > nomorArsip[j + 1]) {
                int t = nomorArsip[j];
                nomorArsip[j] = nomorArsip[j + 1];
                nomorArsip[j + 1] = t;
                totalTukarArsip++;
                adaPertukaran = true;
            }
        }
        if (!adaPertukaran) break;
    }
    cout << "Arsip urut:";
    for (int i = 0; i < jumlahArsip; i++) cout << " " << nomorArsip[i];
    cout << endl << "Putaran: " << putaranDipakai << ", pertukaran: " << totalTukarArsip << endl;
    return 0;
}

Soal 6. Peringkat Saldo Nasabah (Selection Sort Turun)
----------------------------------------------------------------------

Urutkan saldo nasabah dari terbesar dengan selection sort. Pertukaran hanya dilakukan jika posisi terbesar berbeda dari posisi sekarang; hitung berapa kali menukar.

[CONTOH TAMPILAN]
Jumlah nasabah: 6
Saldo (ribu): 500 1200 300 1200 800 100
Urutan saldo: 1200 1200 800 500 300 100
Jumlah pertukaran: 3

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    long long saldoNasabah[100];
    int jumlahNasabah, tukarSaldo = 0;
    cout << "Jumlah nasabah: ";
    cin >> jumlahNasabah;
    cout << "Saldo (ribu): ";
    for (int i = 0; i < jumlahNasabah; i++) cin >> saldoNasabah[i];
    for (int i = 0; i < jumlahNasabah - 1; i++) {
        int posKaya = i;
        for (int j = i + 1; j < jumlahNasabah; j++) if (saldoNasabah[j] > saldoNasabah[posKaya]) posKaya = j;
        if (posKaya != i) {
            long long t = saldoNasabah[i];
            saldoNasabah[i] = saldoNasabah[posKaya];
            saldoNasabah[posKaya] = t;
            tukarSaldo++;
        }
    }
    cout << "Urutan saldo:";
    for (int i = 0; i < jumlahNasabah; i++) cout << " " << saldoNasabah[i];
    cout << endl << "Jumlah pertukaran: " << tukarSaldo << endl;
    return 0;
}

----------------------------------------------------------------------

Pemain kartu menyisipkan setiap kartu baru ke posisi yang tepat di tangannya. Urutkan dengan insertion sort dan tampilkan kartu di tangan setelah tiap sisipan.

[CONTOH TAMPILAN]
Jumlah kartu: 5
Nilai kartu: 7 3 9 1 5
Sisip 3: 3 7
Sisip 9: 3 7 9
Sisip 1: 1 3 7 9
Sisip 5: 1 3 5 7 9

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int kartuTangan[100], jumlahKartuTangan;
    cout << "Jumlah kartu: ";
    cin >> jumlahKartuTangan;
    cout << "Nilai kartu: ";
    for (int i = 0; i < jumlahKartuTangan; i++) cin >> kartuTangan[i];
    for (int i = 1; i < jumlahKartuTangan; i++) {
        int kartuDiambil = kartuTangan[i], j = i - 1;
        while (j >= 0 && kartuTangan[j] > kartuDiambil) {
            kartuTangan[j + 1] = kartuTangan[j];
            j--;
        }
        kartuTangan[j + 1] = kartuDiambil;
        cout << "Sisip " << kartuDiambil << ":";
        for (int k = 0; k <= i; k++) cout << " " << kartuTangan[k];
        cout << endl;
    }
    return 0;
}

Soal 9. Membelah Tumpukan Rapor (Merge Sort)
----------------------------------------------------------------------

Tumpukan rapor dibelah dua terus-menerus sampai tinggal satu, lalu digabung kembali secara urut. Terapkan merge sort rekursif.

[CONTOH TAMPILAN]
Jumlah rapor: 8
Nilai rapor: 38 27 43 3 9 82 10 55
Setelah merge sort: 3 9 10 27 38 43 55 82

[JAWABAN]
#include <iostream>
using namespace std;

void gabungTumpukan(int rapor[], int kiri, int tengah, int kanan) {
    int sementara[100], a = kiri, b = tengah + 1, k = 0;
    while (a <= tengah && b <= kanan) {
        if (rapor[a] <= rapor[b]) sementara[k++] = rapor[a++];
        else sementara[k++] = rapor[b++];
    }
    while (a <= tengah) sementara[k++] = rapor[a++];
    while (b <= kanan) sementara[k++] = rapor[b++];
    for (int i = 0; i < k; i++) rapor[kiri + i] = sementara[i];
}

void belahTumpukan(int rapor[], int kiri, int kanan) {
    if (kiri >= kanan) return;
    int tengah = (kiri + kanan) / 2;
    belahTumpukan(rapor, kiri, tengah);
    belahTumpukan(rapor, tengah + 1, kanan);
    gabungTumpukan(rapor, kiri, tengah, kanan);
}

int main() {
    int nilaiRaporTumpuk[100], jumlahRapor;
    cout << "Jumlah rapor: ";
    cin >> jumlahRapor;
    cout << "Nilai rapor: ";
    for (int i = 0; i < jumlahRapor; i++) cin >> nilaiRaporTumpuk[i];
    belahTumpukan(nilaiRaporTumpuk, 0, jumlahRapor - 1);
    cout << "Setelah merge sort:";
    for (int i = 0; i < jumlahRapor; i++) cout << " " << nilaiRaporTumpuk[i];
    cout << endl;
    return 0;
}

Soal 10. Kapten Pemilih Pivot (Quick Sort)
----------------------------------------------------------------------

Kapten memilih satu pemain sebagai patokan (pivot, elemen terakhir); yang lebih kecil ke kiri, yang lebih besar ke kanan, lalu ulangi pada tiap bagian. Terapkan quick sort.

[CONTOH TAMPILAN]
Jumlah pemain: 7
Kekuatan pemain: 10 80 30 90 40 50 70
Setelah quick sort: 10 30 40 50 70 80 90

[JAWABAN]
#include <iostream>
using namespace std;

int pisahPivot(int barisan[], int awal, int akhir) {
    int pivotKapten = barisan[akhir], batas = awal - 1;
    for (int j = awal; j < akhir; j++) {
        if (barisan[j] < pivotKapten) {
            batas++;
            int t = barisan[batas]; barisan[batas] = barisan[j]; barisan[j] = t;
        }
    }
    int t = barisan[batas + 1]; barisan[batas + 1] = barisan[akhir]; barisan[akhir] = t;
    return batas + 1;
}

void urutCepat(int barisan[], int awal, int akhir) {
    if (awal < akhir) {
        int posPivot = pisahPivot(barisan, awal, akhir);
        urutCepat(barisan, awal, posPivot - 1);
        urutCepat(barisan, posPivot + 1, akhir);
    }
}

int main() {
    int kekuatanPemain[100], jumlahPemainKapten;
    cout << "Jumlah pemain: ";
    cin >> jumlahPemainKapten;
    cout << "Kekuatan pemain: ";
    for (int i = 0; i < jumlahPemainKapten; i++) cin >> kekuatanPemain[i];
    urutCepat(kekuatanPemain, 0, jumlahPemainKapten - 1);
    cout << "Setelah quick sort:";
    for (int i = 0; i < jumlahPemainKapten; i++) cout << " " << kekuatanPemain[i];
    cout << endl;
    return 0;
}

Soal 11. Sensus Umur Cepat (Counting Sort)
----------------------------------------------------------------------

Umur warga berkisar 0–100. Daripada membandingkan, hitung berapa warga untuk tiap umur lalu tuliskan kembali secara urut (counting sort).

[CONTOH TAMPILAN]
Jumlah warga: 10
Umur: 34 5 67 34 12 5 89 5 45 12
Umur terurut: 5 5 5 12 12 34 34 45 67 89

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int umurSensus[100], jumlahDataSensus, hitungUmurSensus[101] = {0};
    cout << "Jumlah warga: ";
    cin >> jumlahDataSensus;
    cout << "Umur: ";
    for (int i = 0; i < jumlahDataSensus; i++) {
        cin >> umurSensus[i];
        hitungUmurSensus[umurSensus[i]]++;
    }
    int posTulis = 0;
    for (int u = 0; u <= 100; u++)
        for (int c = 0; c < hitungUmurSensus[u]; c++) umurSensus[posTulis++] = u;
    cout << "Umur terurut:";
    for (int i = 0; i < jumlahDataSensus; i++) cout << " " << umurSensus[i];

Soal 15. Kunci Anagram Penjaga Gudang Kata
----------------------------------------------------------------------

Urutkan huruf-huruf dalam dua kata (sorting karakter dalam string). Jika hasil urutnya sama, kedua kata adalah anagram.

[CONTOH TAMPILAN]
Dua kata: listen silent
Kunci listen: eilnst
Kunci silent: eilnst
Anagram!

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

string urutHuruf(string kataMentah) {
    for (int i = 1; i < (int)kataMentah.length(); i++) {
        char h = kataMentah[i];
        int j = i - 1;
        while (j >= 0 && kataMentah[j] > h) {
            kataMentah[j + 1] = kataMentah[j];
            j--;
        }
        kataMentah[j + 1] = h;
    }
    return kataMentah;
}

int main() {
    string kataGudangA, kataGudangB;
    cout << "Dua kata: ";
    cin >> kataGudangA >> kataGudangB;
    string kunciA = urutHuruf(kataGudangA), kunciB = urutHuruf(kataGudangB);
    cout << "Kunci " << kataGudangA << ": " << kunciA << endl;
    cout << "Kunci " << kataGudangB << ": " << kunciB << endl;
    if (kunciA == kunciB) cout << "Anagram!" << endl;
    else cout << "Bukan anagram." << endl;
    return 0;
}

#include <string>

    string nama;
    double berat;
    double nilai;
};

    int jenisRempah;
    double kapasitasRansel, nilaiDibawa = 0;
    cout << "Jumlah jenis rempah: ";
    cin >> jenisRempah;
    for (int i = 0; i < jenisRempah; i++) {
        cout << "Rempah " << i + 1 << " (nama berat nilai): ";

            if (gudangRempah[j].nilai / gudangRempah[j].berat < gudangRempah[j + 1].nilai / gudangRempah[j + 1].berat) {

            if (gelasKocok[i] > gelasKocok[i + 1]) { int t = gelasKocok[i]; gelasKocok[i] = gelasKocok[i + 1]; gelasKocok[i + 1] = t; masihTukar = true; }
        ujungAkhir--;
        for (int i = ujungAkhir; i > ujungAwal; i--)

    }
    cout << "Hasil cocktail sort:";
    for (int i = 0; i < jumlahGelas; i++) cout << " " << gelasKocok[i];
    cout << endl;
    return 0;
}

    cout << "Data: ";

                dataShell[j] = dataShell[j - gap];

            dataShell[j] = x;
        }
        cout << "Gap " << gap << ":";
        for (int i = 0; i < jumlahShell; i++) cout << " " << dataShell[i];

Soal 43. Kurcaci Penyusun Pot (Gnome Sort)
----------------------------------------------------------------------

Kurcaci kebun berjalan maju jika dua pot berurutan sudah benar, dan menukar lalu mundur satu langkah jika salah. Terapkan gnome sort dan hitung langkahnya.

[CONTOH TAMPILAN]
Jumlah pot: 6
Nomor pot: 34 2 10 -9 7 1
Pot tersusun: -9 1 2 7 10 34
Langkah kurcaci: 28

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int potKebun[100], jumlahPot, langkahKurcaci = 0;
    cout << "Jumlah pot: ";
    cin >> jumlahPot;
    cout << "Nomor pot: ";
    for (int i = 0; i < jumlahPot; i++) cin >> potKebun[i];
    int posisiKurcaci = 0;
    while (posisiKurcaci < jumlahPot) {
        langkahKurcaci++;
        if (posisiKurcaci == 0 || potKebun[posisiKurcaci - 1] <= potKebun[posisiKurcaci]) {
            posisiKurcaci++;
        } else {
            int t = potKebun[posisiKurcaci];
            potKebun[posisiKurcaci] = potKebun[posisiKurcaci - 1];
            potKebun[posisiKurcaci - 1] = t;
            posisiKurcaci--;
        }
    }
    cout << "Pot tersusun:";
    for (int i = 0; i < jumlahPot; i++) cout << " " << potKebun[i];
    cout << endl << "Langkah kurcaci: " << langkahKurcaci << endl;
    return 0;
}

Soal 44. Lomba Dua Petugas Pengurut
----------------------------------------------------------------------

Salin data yang sama ke dua array. Urutkan satu dengan bubble sort dan satu dengan selection sort, lalu bandingkan jumlah perbandingan dan jumlah pertukaran keduanya.

[CONTOH TAMPILAN]
Jumlah data: 6
Data: 6 5 4 3 2 1
Bubble sort   : 15 perbandingan, 15 pertukaran
Selection sort: 15 perbandingan, 3 pertukaran

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int dataBubble[100], dataSeleksi[100], jumlahDataLomba;
    cout << "Jumlah data: ";
    cin >> jumlahDataLomba;
    cout << "Data: ";
    for (int i = 0; i < jumlahDataLomba; i++) {
        cin >> dataBubble[i];
        dataSeleksi[i] = dataBubble[i];
    }
    int bandingBubble = 0, tukarBubble = 0, bandingSeleksi = 0, tukarSeleksi = 0;
    for (int p = 0; p < jumlahDataLomba - 1; p++)
        for (int j = 0; j < jumlahDataLomba - 1 - p; j++) {
            bandingBubble++;
            if (dataBubble[j] > dataBubble[j + 1]) { int t = dataBubble[j]; dataBubble[j] = dataBubble[j + 1]; dataBubble[j + 1] = t; tukarBubble++; }
        }
    for (int i = 0; i < jumlahDataLomba - 1; i++) {
        int m = i;
        for (int j = i + 1; j < jumlahDataLomba; j++) { bandingSeleksi++; if (dataSeleksi[j] < dataSeleksi[m]) m = j; }
        if (m != i) { int t = dataSeleksi[i]; dataSeleksi[i] = dataSeleksi[m]; dataSeleksi[m] = t; tukarSeleksi++; }
    }
    cout << "Bubble sort   : " << bandingBubble << " perbandingan, " << tukarBubble << " pertukaran" << endl;
    cout << "Selection sort: " << bandingSeleksi << " perbandingan, " << tukarSeleksi << " pertukaran" << endl;
    return 0;
}

[CONTOH TAMPILAN]
5 skor awal (urut turun): 900 750 600 450 300
Skor baru (0 selesai): 700
Skor 700 masuk di peringkat 3
Skor baru (0 selesai): 200
Skor 200 belum masuk papan rekor.
Skor baru (0 selesai): 950
Skor 950 masuk di peringkat 1
Skor baru (0 selesai): 0
Papan rekor akhir: 950 900 750 700 600

[JAWABAN]

using namespace std;

int main() {
    int papanRekor[5];
    cout << "5 skor awal (urut turun): ";
    for (int i = 0; i < 5; i++) cin >> papanRekor[i];
    int skorMasuk;
    cout << "Skor baru (0 selesai): ";
    cin >> skorMasuk;
    while (skorMasuk != 0) {
        if (skorMasuk <= papanRekor[4]) {
            cout << "Skor " << skorMasuk << " belum masuk papan rekor." << endl;
        } else {
            int pos = 4;
            while (pos > 0 && papanRekor[pos - 1] < skorMasuk) {
                papanRekor[pos] = papanRekor[pos - 1];
                pos--;
            }

        }
        cout << "Skor baru (0 selesai): ";
        cin >> skorMasuk;
    }
    cout << "Papan rekor akhir:";
    for (int i = 0; i < 5; i++) cout << " " << papanRekor[i];
    cout << endl;
    return 0;
}

Soal 47. Tabib Median vs Hantu Rentang (pola game Quest IV)
----------------------------------------------------------------------

Tabib Median (HP 40) melawan Hantu Rentang (HP 60). Setiap giliran user memasukkan 3 angka dalam rentang tertentu (mulai 1–10, naik 10 tiap giliran). Urutkan ketiga angka. Jika angka tengah (median) prima, tabib menyerang dengan damage = median % 10 + 10. Jika tidak, hantu menyerang dengan damage = angka terbesar − angka terkecil.

[CONTOH TAMPILAN]
Giliran 1:
Masukkan 3 angka 1-10: 3 5 9
Setelah diurutkan: 3 5 9
Median 5 prima! Tabib menyerang 15 damage!
HP Hantu Rentang: 45

Giliran 2:
Masukkan 3 angka 11-20: 12 15 20
Setelah diurutkan: 12 15 20
Median 15 bukan prima.. Hantu menyerang 8 damage!
HP Tabib Median: 32

Giliran 3:
Masukkan 3 angka 21-30: 21 29 23
Setelah diurutkan: 21 23 29
Median 23 prima! Tabib menyerang 13 damage!
HP Hantu Rentang: 32

Giliran 4:
Masukkan 3 angka 31-40: 31 33 37
Setelah diurutkan: 31 33 37
Median 33 bukan prima.. Hantu menyerang 6 damage!
HP Tabib Median: 26

Giliran 5:
Masukkan 3 angka 41-50: 41 43 49
Setelah diurutkan: 41 43 49
Median 43 prima! Tabib menyerang 13 damage!
HP Hantu Rentang: 19

Giliran 6:
Masukkan 3 angka 51-60: 52 59 53
Setelah diurutkan: 52 53 59
Median 53 prima! Tabib menyerang 13 damage!
HP Hantu Rentang: 6

Giliran 7:
Masukkan 3 angka 61-70: 61 68 67
Setelah diurutkan: 61 67 68
Median 67 prima! Tabib menyerang 17 damage!
HP Hantu Rentang: 0

Hantu Rentang lenyap! Tabib Median menang!

[JAWABAN]
#include <iostream>
using namespace std;

bool primaMedian(int n) {
    if (n < 2) return false;
    for (int d = 2; d * d <= n; d++) if (n % d == 0) return false;
    return true;
}

int main() {
    int hpTabibMedian = 40, hpHantuRentang = 60;
    int giliranMedian = 1, rentangBawahM = 1, rentangAtasM = 10;
    while (hpTabibMedian > 0 && hpHantuRentang > 0) {
        int tigaAngka[3];
        cout << "\nGiliran " << giliranMedian << ":\n";
        cout << "Masukkan 3 angka " << rentangBawahM << "-" << rentangAtasM << ": ";
        bool sahSemua = true;
        for (int i = 0; i < 3; i++) {
            cin >> tigaAngka[i];
            if (tigaAngka[i] < rentangBawahM || tigaAngka[i] > rentangAtasM) sahSemua = false;
        }
        if (!sahSemua) {
            cout << "Ada angka di luar rentang, ulangi!\n";
            continue;
        }
        for (int p = 0; p < 2; p++)
            for (int j = 0; j < 2 - p; j++)
                if (tigaAngka[j] > tigaAngka[j + 1]) { int t = tigaAngka[j]; tigaAngka[j] = tigaAngka[j + 1]; tigaAngka[j + 1] = t; }
        cout << "Setelah diurutkan: " << tigaAngka[0] << " " << tigaAngka[1] << " " << tigaAngka[2] << endl;
        if (primaMedian(tigaAngka[1])) {
            int ramuanMedian = tigaAngka[1] % 10 + 10;
            hpHantuRentang -= ramuanMedian;
            if (hpHantuRentang < 0) hpHantuRentang = 0;
            cout << "Median " << tigaAngka[1] << " prima! Tabib menyerang " << ramuanMedian << " damage!\n";
            cout << "HP Hantu Rentang: " << hpHantuRentang << endl;
        } else {
            int sabetanRentang = tigaAngka[2] - tigaAngka[0];
            hpTabibMedian -= sabetanRentang;
            if (hpTabibMedian < 0) hpTabibMedian = 0;
            cout << "Median " << tigaAngka[1] << " bukan prima.. Hantu menyerang " << sabetanRentang << " damage!\n";
            cout << "HP Tabib Median: " << hpTabibMedian << endl;
        }
        giliranMedian++;
        rentangBawahM += 10;
        rentangAtasM += 10;
    }
    if (hpHantuRentang == 0) cout << "\nHantu Rentang lenyap! Tabib Median menang!\n";
    else cout << "\nTabib Median tumbang... Hantu Rentang menang!\n";
    return 0;
}

Soal 48. Ksatria Urutan vs Penyihir Kacau (pola game Quest IV)
----------------------------------------------------------------------

Ksatria Urutan (HP 40) melawan Penyihir Kacau (HP 60). Setiap giliran user memasukkan bilangan 4 digit dari rentang tertentu (mulai 1000–1999, naik 1000 tiap giliran). Jika digit-digitnya sudah urut naik dari kiri ke kanan (seperti 1239), ksatria menyerang dengan damage 20. Jika belum, penyihir menyerang dengan damage = jumlah pertukaran yang dibutuhkan bubble sort untuk mengurutkan digit-digit tersebut × 2.

[CONTOH TAMPILAN]
Giliran 1:
Masukkan bilangan 1000-1999: 1357
Digit sudah urut! Ksatria menebas 20 damage!
HP Penyihir Kacau: 40

Giliran 2:
Masukkan bilangan 2000-2999: 2431
Butuh 4 pertukaran.. Penyihir menyerang 8 damage!
HP Ksatria Urutan: 32

Giliran 3:
Masukkan bilangan 3000-3999: 3489
Digit sudah urut! Ksatria menebas 20 damage!
HP Penyihir Kacau: 20

Giliran 4:
Masukkan bilangan 4000-4999: 4321
Butuh 6 pertukaran.. Penyihir menyerang 12 damage!
HP Ksatria Urutan: 20

Giliran 5:
Masukkan bilangan 5000-5999: 5678
Digit sudah urut! Ksatria menebas 20 damage!
HP Penyihir Kacau: 0

Penyihir Kacau terurai! Ksatria Urutan menang!

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int hpKsatriaUrut = 40, hpPenyihirKacau = 60;
    int giliranUrut = 1, batasBawahU = 1000, batasAtasU = 1999;
    while (hpKsatriaUrut > 0 && hpPenyihirKacau > 0) {
        int angkaMantra;
        cout << "\nGiliran " << giliranUrut << ":\n";
        cout << "Masukkan bilangan " << batasBawahU << "-" << batasAtasU << ": ";
        cin >> angkaMantra;
        if (angkaMantra < batasBawahU || angkaMantra > batasAtasU) {
            cout << "Di luar rentang, ulangi!\n";
            continue;
        }
        int digitMantra[4], salinMantra = angkaMantra;
        for (int i = 3; i >= 0; i--) { digitMantra[i] = salinMantra % 10; salinMantra /= 10; }
        int tukarMantra = 0;
        for (int p = 0; p < 3; p++)
            for (int j = 0; j < 3 - p; j++)
                if (digitMantra[j] > digitMantra[j + 1]) { int t = digitMantra[j]; digitMantra[j] = digitMantra[j + 1]; digitMantra[j + 1] = t; tukarMantra++; }
        if (tukarMantra == 0) {
            hpPenyihirKacau -= 20;
            if (hpPenyihirKacau < 0) hpPenyihirKacau = 0;
            cout << "Digit sudah urut! Ksatria menebas 20 damage!\n";
            cout << "HP Penyihir Kacau: " << hpPenyihirKacau << endl;
        } else {
            int kutukanKacau = tukarMantra * 2;
            hpKsatriaUrut -= kutukanKacau;
            if (hpKsatriaUrut < 0) hpKsatriaUrut = 0;
            cout << "Butuh " << tukarMantra << " pertukaran.. Penyihir menyerang " << kutukanKacau << " damage!\n";
            cout << "HP Ksatria Urutan: " << hpKsatriaUrut << endl;
        }
        giliranUrut++;
        batasBawahU += 1000;
        batasAtasU += 1000;
    }
    if (hpPenyihirKacau == 0) cout << "\nPenyihir Kacau terurai! Ksatria Urutan menang!\n";
    else cout << "\nKsatria Urutan kalah... Penyihir Kacau menang!\n";
    return 0;
}

----------------------------------------------------------------------
Soal 50. Membagi Tim Seimbang Pola Ular
----------------------------------------------------------------------

Pelatih membagi pemain ke k tim seimbang: urutkan kemampuan dari terbesar, lalu bagikan dengan pola ular (tim 1, 2, ..., k, lalu k, ..., 2, 1, dan seterusnya). Tampilkan anggota dan total kemampuan tiap tim.

[CONTOH TAMPILAN]
Jumlah pemain: 8
Kemampuan pemain: 90 85 80 75 70 65 60 55
Jumlah tim: 2
Tim 1: 90 75 70 55 (total 290)
Tim 2: 85 80 65 60 (total 290)

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int kemampuanPemain[100], jumlahPemainUlar, jumlahTimUlar;
    cout << "Jumlah pemain: ";
    cin >> jumlahPemainUlar;
    cout << "Kemampuan pemain: ";
    for (int i = 0; i < jumlahPemainUlar; i++) cin >> kemampuanPemain[i];
    cout << "Jumlah tim: ";
    cin >> jumlahTimUlar;
    for (int p = 0; p < jumlahPemainUlar - 1; p++)
        for (int j = 0; j < jumlahPemainUlar - 1 - p; j++)
            if (kemampuanPemain[j] < kemampuanPemain[j + 1]) { int t = kemampuanPemain[j]; kemampuanPemain[j] = kemampuanPemain[j + 1]; kemampuanPemain[j + 1] = t; }
    int anggotaTim[10][50], isiTim[10] = {0}, totalTim[10] = {0};
    for (int i = 0; i < jumlahPemainUlar; i++) {
        int putaranUlar = i / jumlahTimUlar, posisiUlar = i % jumlahTimUlar;
        int timTujuan = (putaranUlar % 2 == 0) ? posisiUlar : jumlahTimUlar - 1 - posisiUlar;
        anggotaTim[timTujuan][isiTim[timTujuan]++] = kemampuanPemain[i];
        totalTim[timTujuan] += kemampuanPemain[i];
    }
    for (int t = 0; t < jumlahTimUlar; t++) {
        cout << "Tim " << t + 1 << ":";
        for (int a = 0; a < isiTim[t]; a++) cout << " " << anggotaTim[t][a];
        cout << " (total " << totalTim[t] << ")" << endl;
    }
    return 0;
}
