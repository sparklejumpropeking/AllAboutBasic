Soal 8. Suku Ke-n Kelinci Fibonacci

}

    else cout << nomorBatuCek << " bukan bilangan prima" << endl;

}

Soal 15. Angka Narsis Armstrong
----------------------------------------------------------------------

Angka Armstrong adalah angka yang sama dengan jumlah tiap digitnya dipangkatkan banyak digit (contoh 153 = 1³ + 5³ + 3³). Cek angka yang dimasukkan.

[CONTOH TAMPILAN]
Angka: 153
Jumlah pangkat digit: 153
153 adalah angka Armstrong!

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int angkaNarsis, banyakDigitNarsis = 0, jumlahPangkat = 0;
    cout << "Angka: ";
    cin >> angkaNarsis;
    int salinanNarsis = angkaNarsis;
    while (salinanNarsis > 0) { banyakDigitNarsis++; salinanNarsis /= 10; }
    salinanNarsis = angkaNarsis;
    while (salinanNarsis > 0) {
        int digitNarsis = salinanNarsis % 10, hasilPangkat = 1;
        for (int kaliPangkat = 0; kaliPangkat < banyakDigitNarsis; kaliPangkat++) hasilPangkat *= digitNarsis;
        jumlahPangkat += hasilPangkat;
        salinanNarsis /= 10;
    }
    cout << "Jumlah pangkat digit: " << jumlahPangkat << endl;
    if (jumlahPangkat == angkaNarsis) cout << angkaNarsis << " adalah angka Armstrong!" << endl;
    else cout << angkaNarsis << " bukan angka Armstrong." << endl;
    return 0;
}

Soal 40. ATM Koperasi Desa
----------------------------------------------------------------------

Buat ATM sederhana dengan do-while dan switch: 1 cek saldo, 2 setor, 3 tarik (tolak jika saldo kurang), 4 keluar. Saldo awal Rp500.000.

[CONTOH TAMPILAN]
1.Cek saldo 2.Setor 3.Tarik 4.Keluar
Pilih: 2
Nominal setor: Rp150000
Setor berhasil.

1.Cek saldo 2.Setor 3.Tarik 4.Keluar
Pilih: 3
Nominal tarik: Rp800000
Saldo tidak cukup!

1.Cek saldo 2.Setor 3.Tarik 4.Keluar
Pilih: 3
Nominal tarik: Rp200000
Tarik berhasil.

1.Cek saldo 2.Setor 3.Tarik 4.Keluar
Pilih: 1
Saldo: Rp450000

1.Cek saldo 2.Setor 3.Tarik 4.Keluar
Pilih: 4
Terima kasih telah menggunakan ATM Koperasi.

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    long long saldoKoperasi = 500000, nominalTransaksi;
    int menuKoperasi;
    do {
        cout << "\n1.Cek saldo 2.Setor 3.Tarik 4.Keluar\nPilih: ";
        cin >> menuKoperasi;
        switch (menuKoperasi) {
            case 1: cout << "Saldo: Rp" << saldoKoperasi << endl; break;
            case 2:
                cout << "Nominal setor: Rp";
                cin >> nominalTransaksi;
                saldoKoperasi += nominalTransaksi;
                cout << "Setor berhasil." << endl;
                break;
            case 3:
                cout << "Nominal tarik: Rp";
                cin >> nominalTransaksi;
                if (nominalTransaksi > saldoKoperasi) cout << "Saldo tidak cukup!" << endl;
                else { saldoKoperasi -= nominalTransaksi; cout << "Tarik berhasil." << endl; }
                break;
            case 4: cout << "Terima kasih telah menggunakan ATM Koperasi." << endl; break;
            default: cout << "Menu tidak tersedia." << endl;
        }
    } while (menuKoperasi != 4);
    return 0;
}

Soal 49. Lift Gedung Pencakar Langit
----------------------------------------------------------------------

Lift mulai dari lantai 1. Masukkan lantai tujuan berulang kali (0 untuk berhenti). Lantai di luar 1–30 ditolak. Hitung total lantai yang dilalui lift.

[CONTOH TAMPILAN]
Lantai tujuan (0 = selesai): 10
Lift tiba di lantai 10
Lantai tujuan (0 = selesai): 4
Lift tiba di lantai 4
Lantai tujuan (0 = selesai): 35
Lantai tidak ada!
Lantai tujuan (0 = selesai): 20
Lift tiba di lantai 20
Lantai tujuan (0 = selesai): 0
Total lantai yang dilalui: 31

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int lantaiLift = 1, tujuanLift, totalLantaiDilalui = 0;
    while (true) {
        cout << "Lantai tujuan (0 = selesai): ";
        cin >> tujuanLift;
        if (tujuanLift == 0) break;
        if (tujuanLift < 1 || tujuanLift > 30) {
            cout << "Lantai tidak ada!" << endl;
            continue;
        }
        int jarakLift = tujuanLift - lantaiLift;
        if (jarakLift < 0) jarakLift = -jarakLift;
        totalLantaiDilalui += jarakLift;
        lantaiLift = tujuanLift;
        cout << "Lift tiba di lantai " << lantaiLift << endl;
    }
    cout << "Total lantai yang dilalui: " << totalLantaiDilalui << endl;
    return 0;
}

Soal 50. Rapor Kelas Tanpa Array
----------------------------------------------------------------------

Masukkan nilai n siswa. Hitung rata-rata, jumlah yang lulus (≥ 75), dan jumlah yang remedial.

[CONTOH TAMPILAN]
Jumlah siswa: 5
Nilai siswa: 80 65 90 74 88
Rata-rata kelas : 79.4
Lulus           : 3
Remedial        : 2

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int siswaKelas, nilaiSiswa, siswaLulus = 0, siswaRemedial = 0;
    double totalNilaiKelas = 0;
    cout << "Jumlah siswa: ";
    cin >> siswaKelas;
    cout << "Nilai siswa: ";
    for (int urutSiswa = 0; urutSiswa < siswaKelas; urutSiswa++) {
        cin >> nilaiSiswa;
        totalNilaiKelas += nilaiSiswa;
        if (nilaiSiswa >= 75) siswaLulus++;
        else siswaRemedial++;
    }
    cout << "Rata-rata kelas : " << totalNilaiKelas / siswaKelas << endl;
    cout << "Lulus           : " << siswaLulus << endl;
    cout << "Remedial        : " << siswaRemedial << endl;
    return 0;
}
