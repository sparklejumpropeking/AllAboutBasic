Nilai menari: 88 75 92 69

}

Soal 5. Murid di Atas Rata-rata

[CONTOH TAMPILAN]
Jumlah murid: 6
Nilai: 70 85 60 90 75 80

Nilai di atas rata-rata: 85 90 80

[JAWABAN]

using namespace std;

int main() {
    int nilaiKelasArr[100], banyakMuridArr;
    double totalNilaiArr = 0;
    cout << "Jumlah murid: ";
    cin >> banyakMuridArr;
    cout << "Nilai: ";
    for (int muridIdx = 0; muridIdx < banyakMuridArr; muridIdx++) {
        cin >> nilaiKelasArr[muridIdx];
        totalNilaiArr += nilaiKelasArr[muridIdx];
    }
    double rataKelasArr = totalNilaiArr / banyakMuridArr;
    int diAtasRata = 0;
    cout << "Rata-rata kelas: " << rataKelasArr << endl;
    cout << "Nilai di atas rata-rata:";
    for (int muridIdx = 0; muridIdx < banyakMuridArr; muridIdx++) {
        if (nilaiKelasArr[muridIdx] > rataKelasArr) {
            cout << " " << nilaiKelasArr[muridIdx];
            diAtasRata++;
        }
    }
    cout << endl << "Banyaknya: " << diAtasRata << " murid" << endl;

Simpan nama dan harga 5 menu dalam array yang langsung diinisialisasi. Pembeli memasukkan nomor menu berkali-kali (0 untuk selesai), tampilkan total bayarnya.

[CONTOH TAMPILAN]
1. Nasi - Rp5000
2. Tempe Orek - Rp4000
3. Telur Balado - Rp6000
4. Sayur Asem - Rp3000
5. Es Teh - Rp3000
Pilih menu (0 = selesai): 1
+ Nasi
Pilih menu (0 = selesai): 3
+ Telur Balado
Pilih menu (0 = selesai): 3
+ Telur Balado
Pilih menu (0 = selesai): 5
+ Es Teh

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

int main() {
    string namaLaukWarteg[5] = {"Nasi", "Tempe Orek", "Telur Balado", "Sayur Asem", "Es Teh"};
    int hargaLaukWarteg[5] = {5000, 4000, 6000, 3000, 3000};
    int pilihanLauk, totalWarteg = 0;
    for (int nomorLauk = 0; nomorLauk < 5; nomorLauk++)
        cout << nomorLauk + 1 << ". " << namaLaukWarteg[nomorLauk] << " - Rp" << hargaLaukWarteg[nomorLauk] << endl;
    cout << "Pilih menu (0 = selesai): ";
    cin >> pilihanLauk;
    while (pilihanLauk != 0) {
        if (pilihanLauk >= 1 && pilihanLauk <= 5) {
            totalWarteg += hargaLaukWarteg[pilihanLauk - 1];
            cout << "+ " << namaLaukWarteg[pilihanLauk - 1] << endl;
        } else {
            cout << "Menu tidak ada!" << endl;
        }
        cout << "Pilih menu (0 = selesai): ";
        cin >> pilihanLauk;
    }
    cout << "Total bayar: Rp" << totalWarteg << endl;
    return 0;
}

using namespace std;

using namespace std;

Soal 16. Juara Kedua Lomba Lari
----------------------------------------------------------------------

Cari waktu tercepat kedua (nilai terkecil kedua yang berbeda dari yang tercepat) tanpa mengurutkan array.

[CONTOH TAMPILAN]
Jumlah pelari: 6
Waktu tiap pelari (detik): 58 61 55 70 55 57
Juara 1: 55 detik
Juara 2: 57 detik

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int waktuLari[100], pesertaLari;
    cout << "Jumlah pelari: ";
    cin >> pesertaLari;
    cout << "Waktu tiap pelari (detik): ";
    for (int p = 0; p < pesertaLari; p++) cin >> waktuLari[p];
    int tercepatSatu = waktuLari[0], tercepatDua = -1;
    for (int p = 1; p < pesertaLari; p++) if (waktuLari[p] < tercepatSatu) tercepatSatu = waktuLari[p];
    for (int p = 0; p < pesertaLari; p++) {
        if (waktuLari[p] != tercepatSatu && (tercepatDua == -1 || waktuLari[p] < tercepatDua)) tercepatDua = waktuLari[p];
    }
    cout << "Juara 1: " << tercepatSatu << " detik" << endl;
    if (tercepatDua == -1) cout << "Tidak ada juara 2 (semua waktu sama)" << endl;
    else cout << "Juara 2: " << tercepatDua << " detik" << endl;
    return 0;
}

Soal 17. Lonjakan Suhu Terbesar
----------------------------------------------------------------------

Simpan suhu harian. Cari kenaikan suhu terbesar dari satu hari ke hari berikutnya dan di hari ke berapa itu terjadi.

[CONTOH TAMPILAN]
Jumlah hari: 7
Suhu tiap hari: 28 30 27 33 34 29 35
Kenaikan terbesar 6 derajat, terjadi di hari ke-4

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int suhuHarianKota[100], hariPencatatan;
    cout << "Jumlah hari: ";
    cin >> hariPencatatan;
    cout << "Suhu tiap hari: ";
    for (int h = 0; h < hariPencatatan; h++) cin >> suhuHarianKota[h];
    int lonjakanMaks = suhuHarianKota[1] - suhuHarianKota[0], hariLonjakan = 2;
    for (int h = 2; h < hariPencatatan; h++) {
        if (suhuHarianKota[h] - suhuHarianKota[h - 1] > lonjakanMaks) {
            lonjakanMaks = suhuHarianKota[h] - suhuHarianKota[h - 1];
            hariLonjakan = h + 1;
        }
    }
    cout << "Kenaikan terbesar " << lonjakanMaks << " derajat, terjadi di hari ke-" << hariLonjakan << endl;
    return 0;
}

    int tglHari, blnHari, thnHari, urutanHari = 0;
    cout << "Tanggal bulan tahun: ";
    cin >> tglHari >> blnHari >> thnHari;
    if ((thnHari % 4 == 0 && thnHari % 100 != 0) || thnHari % 400 == 0) hariPerBulan[1] = 29;
    for (int b = 0; b < blnHari - 1; b++) urutanHari += hariPerBulan[b];

----------------------------------------------------------------------
Soal 48. Antrean Loket Kelurahan (Queue dengan Array)
----------------------------------------------------------------------

Simulasikan antrean memakai array. Perintah: 1 x = warga bernomor x datang (masuk belakang), 2 = warga paling depan dilayani, 3 = tampilkan antrean, 0 = tutup loket.

[CONTOH TAMPILAN]
Perintah: 1 101
Warga 101 masuk antrean
Perintah: 1 102
Warga 102 masuk antrean
Perintah: 1 103
Warga 103 masuk antrean
Perintah: 2
Warga 101 dilayani
Perintah: 3
Antrean: 102 103
Perintah: 1 104
Warga 104 masuk antrean
Perintah: 3
Antrean: 102 103 104
Perintah: 0
Loket tutup.

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int barisLoket[100], kepalaAntre = 0, ekorAntre = 0, perintahLoket, nomorWarga;
    cout << "Perintah: ";
    cin >> perintahLoket;
    while (perintahLoket != 0) {
        if (perintahLoket == 1) {
            cin >> nomorWarga;
            barisLoket[ekorAntre++] = nomorWarga;
            cout << "Warga " << nomorWarga << " masuk antrean" << endl;
        } else if (perintahLoket == 2) {
            if (kepalaAntre == ekorAntre) cout << "Antrean kosong" << endl;
            else cout << "Warga " << barisLoket[kepalaAntre++] << " dilayani" << endl;
        } else if (perintahLoket == 3) {
            cout << "Antrean:";
            for (int a = kepalaAntre; a < ekorAntre; a++) cout << " " << barisLoket[a];
            cout << endl;
        }
        cout << "Perintah: ";
        cin >> perintahLoket;
    }
    cout << "Loket tutup." << endl;
    return 0;
}

----------------------------------------------------------------------
Soal 49. Tumpukan Piring Warung (Stack dengan Array)
----------------------------------------------------------------------

Tumpukan piring hanya bisa diambil dari atas. Perintah: 1 x = taruh piring bernomor x di atas, 2 = ambil piring paling atas, 3 = lihat tumpukan dari atas, 0 = selesai. Kapasitas 5 piring.

[CONTOH TAMPILAN]
Aksi: 1 11
Piring 11 ditaruh
Aksi: 1 12
Piring 12 ditaruh
Aksi: 1 13
Piring 13 ditaruh
Aksi: 2
Piring 13 diambil
Aksi: 3
Tumpukan (atas ke bawah): 12 11
Aksi: 0

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int tumpukanPiring[5], puncakPiring = 0, aksiPiring, nomorPiring;
    cout << "Aksi: ";
    cin >> aksiPiring;
    while (aksiPiring != 0) {
        if (aksiPiring == 1) {
            cin >> nomorPiring;
            if (puncakPiring == 5) cout << "Tumpukan penuh!" << endl;
            else {
                tumpukanPiring[puncakPiring++] = nomorPiring;
                cout << "Piring " << nomorPiring << " ditaruh" << endl;
            }
        } else if (aksiPiring == 2) {
            if (puncakPiring == 0) cout << "Tidak ada piring!" << endl;
            else cout << "Piring " << tumpukanPiring[--puncakPiring] << " diambil" << endl;
        } else if (aksiPiring == 3) {
            cout << "Tumpukan (atas ke bawah):";
            for (int p = puncakPiring - 1; p >= 0; p--) cout << " " << tumpukanPiring[p];
            cout << endl;
        }
        cout << "Aksi: ";
        cin >> aksiPiring;
    }
    return 0;
}

----------------------------------------------------------------------
Soal 50. Petak Parkir Pasar Pagi
----------------------------------------------------------------------

Pasar punya 6 petak parkir (array, 0 = kosong). Perintah: 1 plat = mobil masuk ke petak kosong pertama, 2 nomorPetak = mobil di petak itu keluar, 3 = tampilkan isi semua petak, 0 = selesai.

[CONTOH TAMPILAN]
Aksi: 1 4512
Mobil 4512 parkir di petak 1
Aksi: 1 7781
Mobil 7781 parkir di petak 2
Aksi: 1 3090
Mobil 3090 parkir di petak 3
Aksi: 2 2
Mobil 7781 keluar dari petak 2
Aksi: 1 6604
Mobil 6604 parkir di petak 2
Aksi: 3
Petak 1: 4512
Petak 2: 6604
Petak 3: 3090
Petak 4: kosong
Petak 5: kosong
Petak 6: kosong
Aksi: 0

[JAWABAN]
#include <iostream>
using namespace std;

int main() {
    int petakParkirPasar[6] = {0}, aksiParkir, dataParkir;
    cout << "Aksi: ";
    cin >> aksiParkir;
    while (aksiParkir != 0) {
        if (aksiParkir == 1) {
            cin >> dataParkir;
            int petakKosong = -1;
            for (int p = 0; p < 6; p++) if (petakParkirPasar[p] == 0) { petakKosong = p; break; }
            if (petakKosong == -1) cout << "Parkiran penuh!" << endl;
            else {
                petakParkirPasar[petakKosong] = dataParkir;
                cout << "Mobil " << dataParkir << " parkir di petak " << petakKosong + 1 << endl;
            }
        } else if (aksiParkir == 2) {
            cin >> dataParkir;
            if (dataParkir < 1 || dataParkir > 6 || petakParkirPasar[dataParkir - 1] == 0) cout << "Petak itu kosong!" << endl;
            else {
                cout << "Mobil " << petakParkirPasar[dataParkir - 1] << " keluar dari petak " << dataParkir << endl;
                petakParkirPasar[dataParkir - 1] = 0;
            }
        } else if (aksiParkir == 3) {
            for (int p = 0; p < 6; p++) {
                cout << "Petak " << p + 1 << ": ";
                if (petakParkirPasar[p] == 0) cout << "kosong" << endl;
                else cout << petakParkirPasar[p] << endl;
            }
        }
        cout << "Aksi: ";
        cin >> aksiParkir;
    }
    return 0;
}
