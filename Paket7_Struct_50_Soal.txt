Soal 3. Kue Pecahan Nenek
----------------------------------------------------------------------

Buat struct Pecahan (pembilang, penyebut). Jumlahkan dua pecahan kue dan sederhanakan hasilnya memakai FPB.

[CONTOH TAMPILAN]
Bagian kue anak 1 (pembilang penyebut): 1 4
Bagian kue anak 2 (pembilang penyebut): 2 6
Total kue yang dimakan: 7/12

[JAWABAN]
#include <iostream>
using namespace std;

struct Pecahan {
    int pembilang;
    int penyebut;
};

int fpbKue(int a, int b) {
    while (b != 0) { int s = a % b; a = b; b = s; }
    return a;
}

int main() {
    Pecahan kueAnakSatu, kueAnakDua, totalKue;
    cout << "Bagian kue anak 1 (pembilang penyebut): ";
    cin >> kueAnakSatu.pembilang >> kueAnakSatu.penyebut;
    cout << "Bagian kue anak 2 (pembilang penyebut): ";
    cin >> kueAnakDua.pembilang >> kueAnakDua.penyebut;
    totalKue.pembilang = kueAnakSatu.pembilang * kueAnakDua.penyebut + kueAnakDua.pembilang * kueAnakSatu.penyebut;
    totalKue.penyebut = kueAnakSatu.penyebut * kueAnakDua.penyebut;
    int pembagiKue = fpbKue(totalKue.pembilang, totalKue.penyebut);
    totalKue.pembilang /= pembagiKue;
    totalKue.penyebut /= pembagiKue;
    cout << "Total kue yang dimakan: " << totalKue.pembilang << "/" << totalKue.penyebut << endl;
    return 0;
}

Tanggal (hari bulan tahun): 17 8 1945

#include <string>
using namespace std;

    int hari;
    int bulan;
    int tahun;
};

    cin >> tanggalSurat.hari >> tanggalSurat.bulan >> tanggalSurat.tahun;
    if (tanggalSurat.bulan < 1 || tanggalSurat.bulan > 12) {
        cout << "Bulan tidak valid!" << endl;

    return 0;

Soal 9. Rekening Koperasi Petani
----------------------------------------------------------------------

Buat struct Rekening (nama pemilik, saldo). Buat prosedur setor dan fungsi bool tarik yang menerima struct by reference sehingga saldonya benar-benar berubah.

[CONTOH TAMPILAN]
Nama pemilik: Pak_Darmo
Saldo awal: 1000000
Setor: 250000
Saldo sekarang: Rp1250000
Tarik: 1500000
Gagal, saldo tidak cukup.
Tarik: 400000
Berhasil. Saldo: Rp850000

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

struct Rekening {
    string pemilik;
    long long saldo;
};

void setorTabungan(Rekening &rekTani, long long jumlahSetor) {
    rekTani.saldo += jumlahSetor;
}

bool tarikTabungan(Rekening &rekTani, long long jumlahTarik) {
    if (jumlahTarik > rekTani.saldo) return false;
    rekTani.saldo -= jumlahTarik;
    return true;
}

int main() {
    Rekening rekeningKoperasi;
    long long nominalSetor, tarikPertama, tarikKedua;
    cout << "Nama pemilik: ";
    cin >> rekeningKoperasi.pemilik;
    cout << "Saldo awal: ";
    cin >> rekeningKoperasi.saldo;
    cout << "Setor: ";
    cin >> nominalSetor;
    setorTabungan(rekeningKoperasi, nominalSetor);
    cout << "Saldo sekarang: Rp" << rekeningKoperasi.saldo << endl;
    cout << "Tarik: ";
    cin >> tarikPertama;
    if (tarikTabungan(rekeningKoperasi, tarikPertama)) cout << "Berhasil. Saldo: Rp" << rekeningKoperasi.saldo << endl;
    else cout << "Gagal, saldo tidak cukup." << endl;
    cout << "Tarik: ";
    cin >> tarikKedua;
    if (tarikTabungan(rekeningKoperasi, tarikKedua)) cout << "Berhasil. Saldo: Rp" << rekeningKoperasi.saldo << endl;
    else cout << "Gagal, saldo tidak cukup." << endl;
    return 0;
}

[JAWABAN]

Soal 13. Jangkauan Menara Sinyal
----------------------------------------------------------------------

Buat struct Titik dan struct Menara (pusat bertipe Titik, jangkauan radius). Cek apakah rumah warga berada dalam jangkauan sinyal.

[CONTOH TAMPILAN]
Posisi menara (x y) dan jangkauan: 0 0 5
Posisi rumah (x y): 3 4
Rumah mendapat sinyal!

[JAWABAN]
#include <iostream>
using namespace std;

struct Titik {
    double x, y;
};

struct Menara {
    Titik pusat;
    double radius;
};

int main() {
    Menara menaraSinyal;
    Titik rumahWarga;
    cout << "Posisi menara (x y) dan jangkauan: ";
    cin >> menaraSinyal.pusat.x >> menaraSinyal.pusat.y >> menaraSinyal.radius;
    cout << "Posisi rumah (x y): ";
    cin >> rumahWarga.x >> rumahWarga.y;
    double selisihXR = rumahWarga.x - menaraSinyal.pusat.x;
    double selisihYR = rumahWarga.y - menaraSinyal.pusat.y;
    if (selisihXR * selisihXR + selisihYR * selisihYR <= menaraSinyal.radius * menaraSinyal.radius)
        cout << "Rumah mendapat sinyal!" << endl;
    else
        cout << "Rumah di luar jangkauan sinyal." << endl;
    return 0;
}

Soal 14. Kalkulator Bilangan Kompleks Insinyur
----------------------------------------------------------------------

Buat struct Kompleks (real, imajiner) dan fungsi tambah serta kali yang mengembalikan struct. Rumus kali: (a+bi)(c+di) = (ac−bd) + (ad+bc)i.

[CONTOH TAMPILAN]
Bilangan 1 (real imajiner): 3 2
Bilangan 2 (real imajiner): 1 4
Jumlah : 4 + 6i
Kali   : -5 + 14i

[JAWABAN]
#include <iostream>
using namespace std;

struct Kompleks {
    double real, imaj;
};

Kompleks tambahKompleks(Kompleks p, Kompleks q) {
    Kompleks hasilTambah = {p.real + q.real, p.imaj + q.imaj};
    return hasilTambah;
}

Kompleks kaliKompleks(Kompleks p, Kompleks q) {
    Kompleks hasilKali = {p.real * q.real - p.imaj * q.imaj, p.real * q.imaj + p.imaj * q.real};
    return hasilKali;
}

void cetakKompleks(Kompleks k) {
    cout << k.real << (k.imaj < 0 ? " - " : " + ") << (k.imaj < 0 ? -k.imaj : k.imaj) << "i" << endl;
}

int main() {
    Kompleks arusSatu, arusDua;
    cout << "Bilangan 1 (real imajiner): ";
    cin >> arusSatu.real >> arusSatu.imaj;
    cout << "Bilangan 2 (real imajiner): ";
    cin >> arusDua.real >> arusDua.imaj;
    cout << "Jumlah : ";
    cetakKompleks(tambahKompleks(arusSatu, arusDua));
    cout << "Kali   : ";
    cetakKompleks(kaliKompleks(arusSatu, arusDua));
    return 0;
}

Soal 15. Jam Weker Penjaga Mercusuar
----------------------------------------------------------------------

Buat struct Jam (jam, menit, detik) dan prosedur tambahDetik yang menambah sejumlah detik ke jam (by reference) dengan perpindahan otomatis ke menit, jam, dan hari berikutnya. Tampilkan dalam format HH:MM:SS.

[CONTOH TAMPILAN]
Jam sekarang (jam menit detik): 23 58 30
Tambah berapa detik: 150
Jam menjadi: 00:01:00

[JAWABAN]
#include <iostream>
#include <iomanip>
using namespace std;

struct Jam {
    int jam, menit, detik;
};

void tambahDetik(Jam &wekerMercu, int detikTambah) {
    int totalDetikWeker = wekerMercu.jam * 3600 + wekerMercu.menit * 60 + wekerMercu.detik + detikTambah;
    totalDetikWeker %= 24 * 3600;
    wekerMercu.jam = totalDetikWeker / 3600;
    wekerMercu.menit = totalDetikWeker % 3600 / 60;
    wekerMercu.detik = totalDetikWeker % 60;
}

void tampilJam(Jam wekerMercu) {
    cout << setfill('0') << setw(2) << wekerMercu.jam << ":" << setw(2) << wekerMercu.menit << ":" << setw(2) << wekerMercu.detik << endl;
}

int main() {
    Jam wekerPenjaga;
    int detikDitambah;
    cout << "Jam sekarang (jam menit detik): ";
    cin >> wekerPenjaga.jam >> wekerPenjaga.menit >> wekerPenjaga.detik;
    cout << "Tambah berapa detik: ";
    cin >> detikDitambah;
    tambahDetik(wekerPenjaga, detikDitambah);
    cout << "Jam menjadi: ";
    tampilJam(wekerPenjaga);
    return 0;

Buat struct Karyawan (nama, gaji pokok, jam lembur, jumlah absen). Gaji bersih = gaji pokok + lembur × Rp30.000 − absen × Rp100.000. Cetak slip gaji lewat prosedur yang menerima struct.

[CONTOH TAMPILAN]
Nama, gaji pokok, jam lembur, absen: Sumarni 4000000 12 2
===== SLIP GAJI =====
Nama         : Sumarni
Gaji pokok   : Rp4000000
Lembur       : Rp360000
Potongan     : Rp200000
Gaji bersih  : Rp4160000

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

struct Karyawan {
    string nama;
    long long gajiPokok;
    int jamLembur;
    int absen;
};

void cetakSlip(Karyawan kry) {
    long long uangLemburKry = kry.jamLembur * 30000LL;
    long long potonganAbsen = kry.absen * 100000LL;
    cout << "===== SLIP GAJI =====" << endl;
    cout << "Nama         : " << kry.nama << endl;
    cout << "Gaji pokok   : Rp" << kry.gajiPokok << endl;
    cout << "Lembur       : Rp" << uangLemburKry << endl;
    cout << "Potongan     : Rp" << potonganAbsen << endl;
    cout << "Gaji bersih  : Rp" << kry.gajiPokok + uangLemburKry - potonganAbsen << endl;
}

int main() {
    Karyawan buruhTekstil;
    cout << "Nama, gaji pokok, jam lembur, absen: ";
    cin >> buruhTekstil.nama >> buruhTekstil.gajiPokok >> buruhTekstil.jamLembur >> buruhTekstil.absen;
    cetakSlip(buruhTekstil);
    return 0;
}

[CONTOH TAMPILAN]
Jumlah buku: 3
Buku 1 (kode judul): B01 Atlas
Buku 2 (kode judul): B02 Kamus
Buku 3 (kode judul): B03 Ensiklopedia

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

struct BukuPinjam {
    string kode;
    string judul;
    bool dipinjam;
};

    BukuPinjam mobilPustaka[20];
    int koleksiBuku;
    cout << "Jumlah buku: ";
    cin >> koleksiBuku;
    for (int i = 0; i < koleksiBuku; i++) {
        cout << "Buku " << i + 1 << " (kode judul): ";

        cin >> kodePerintah;

            if (mobilPustaka[posBuku].dipinjam) cout << mobilPustaka[posBuku].judul << " sedang dipinjam orang lain." << endl;
            else { mobilPustaka[posBuku].dipinjam = true; cout << mobilPustaka[posBuku].judul << " berhasil dipinjam." << endl; }

            if (!mobilPustaka[posBuku].dipinjam) cout << mobilPustaka[posBuku].judul << " memang tidak sedang dipinjam." << endl;
            else { mobilPustaka[posBuku].dipinjam = false; cout << mobilPustaka[posBuku].judul << " sudah dikembalikan." << endl; }
        }
        cout << "Perintah (P/K kode, S selesai): ";
        cin >> perintahPustaka;
    }
    for (int i = 0; i < koleksiBuku; i++)
        cout << mobilPustaka[i].kode << " " << mobilPustaka[i].judul << ": " << (mobilPustaka[i].dipinjam ? "dipinjam" : "tersedia") << endl;
    return 0;
}

Soal 33. Buku Tabungan dengan Riwayat
----------------------------------------------------------------------

Buat struct Tabungan berisi nama, saldo, serta array riwayat transaksi (nominal bertanda: + setor, − tarik) dan jumlah transaksinya. Masukkan transaksi sampai 0, lalu cetak mutasi rekening.

[CONTOH TAMPILAN]
Nama dan saldo awal: Wahyu 500000
Transaksi (+setor / -tarik, 0 selesai): 200000
Transaksi (+setor / -tarik, 0 selesai): -100000
Transaksi (+setor / -tarik, 0 selesai): -900000
Ditolak, saldo tidak cukup.
Transaksi (+setor / -tarik, 0 selesai): 50000
Transaksi (+setor / -tarik, 0 selesai): 0
== Mutasi Wahyu ==
1. Setor 200000
2. Tarik 100000
3. Setor 50000
Saldo akhir: Rp650000

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

struct Tabungan {
    string nama;
    long long saldo;
    long long riwayat[50];
    int banyakRiwayat;
};

int main() {
    Tabungan bukuTabungan;
    bukuTabungan.banyakRiwayat = 0;
    cout << "Nama dan saldo awal: ";
    cin >> bukuTabungan.nama >> bukuTabungan.saldo;
    long long transaksiBaru;
    cout << "Transaksi (+setor / -tarik, 0 selesai): ";
    cin >> transaksiBaru;
    while (transaksiBaru != 0) {
        if (bukuTabungan.saldo + transaksiBaru < 0) cout << "Ditolak, saldo tidak cukup." << endl;
        else {
            bukuTabungan.saldo += transaksiBaru;
            bukuTabungan.riwayat[bukuTabungan.banyakRiwayat++] = transaksiBaru;
        }
        cout << "Transaksi (+setor / -tarik, 0 selesai): ";
        cin >> transaksiBaru;
    }
    cout << "== Mutasi " << bukuTabungan.nama << " ==" << endl;
    for (int i = 0; i < bukuTabungan.banyakRiwayat; i++)
        cout << i + 1 << ". " << (bukuTabungan.riwayat[i] > 0 ? "Setor " : "Tarik ") << (bukuTabungan.riwayat[i] > 0 ? bukuTabungan.riwayat[i] : -bukuTabungan.riwayat[i]) << endl;
    cout << "Saldo akhir: Rp" << bukuTabungan.saldo << endl;
    return 0;
}

[CONTOH TAMPILAN]
Perintah (1 nomor=datang, 2=layani, 0=tutup): 1 11
Pasien 11 mengantre
Perintah (1 nomor=datang, 2=layani, 0=tutup): 1 12
Pasien 12 mengantre
Perintah (1 nomor=datang, 2=layani, 0=tutup): 2
Memanggil pasien 11

Perintah (1 nomor=datang, 2=layani, 0=tutup): 2

Perintah (1 nomor=datang, 2=layani, 0=tutup): 2

[JAWABAN]
#include <iostream>
using namespace std;

struct Antrean {
    int nomor[100];
    int depan;
    int belakang;
};

void masukAntrean(Antrean &aq, int nomorPasien) {
    aq.nomor[aq.belakang++] = nomorPasien;
}

int layaniAntrean(Antrean &aq) {
    if (aq.depan == aq.belakang) return -1;
    return aq.nomor[aq.depan++];
}

int main() {
    Antrean antreanPuskesmas;
    antreanPuskesmas.depan = 0;
    antreanPuskesmas.belakang = 0;
    int perintahPuskesmas, nomorDatang;
    cout << "Perintah (1 nomor=datang, 2=layani, 0=tutup): ";
    cin >> perintahPuskesmas;
    while (perintahPuskesmas != 0) {
        if (perintahPuskesmas == 1) {
            cin >> nomorDatang;
            masukAntrean(antreanPuskesmas, nomorDatang);
            cout << "Pasien " << nomorDatang << " mengantre" << endl;
        } else if (perintahPuskesmas == 2) {
            int dilayani = layaniAntrean(antreanPuskesmas);
            if (dilayani == -1) cout << "Tidak ada pasien" << endl;
            else cout << "Memanggil pasien " << dilayani << endl;
        }
        cout << "Perintah (1 nomor=datang, 2=layani, 0=tutup): ";
        cin >> perintahPuskesmas;
    }
    return 0;
}

[CONTOH TAMPILAN]
Perintah: T Izin_Usaha
Izin_Usaha ditaruh di atas tumpukan
Perintah: T KTP_Baru
KTP_Baru ditaruh di atas tumpukan
Perintah: T Surat_Pindah

Perintah: T Akta_Lahir
Akta_Lahir ditaruh di atas tumpukan
Perintah: A

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

struct Tumpukan {
    string berkas[30];
    int puncak;
};

void taruhBerkas(Tumpukan &tp, string judulBerkas) {
    tp.berkas[tp.puncak++] = judulBerkas;
}

string ambilBerkas(Tumpukan &tp) {
    if (tp.puncak == 0) return "";
    return tp.berkas[--tp.puncak];
}

int main() {
    Tumpukan mejaLurah;
    mejaLurah.puncak = 0;
    string perintahLurah;
    cout << "Perintah: ";
    cin >> perintahLurah;
    while (perintahLurah != "0") {
        if (perintahLurah == "T") {
            string judulMasuk;
            cin >> judulMasuk;
            taruhBerkas(mejaLurah, judulMasuk);
            cout << judulMasuk << " ditaruh di atas tumpukan" << endl;
        } else if (perintahLurah == "A") {
            string diproses = ambilBerkas(mejaLurah);
            if (diproses == "") cout << "Meja kosong!" << endl;
            else cout << "Lurah memproses " << diproses << endl;
        }
        cout << "Perintah: ";
        cin >> perintahLurah;
    }
    cout << "Sisa berkas di meja: " << mejaLurah.puncak << endl;
    return 0;
}

Soal 37. Harga Promo Toko Elektronik
----------------------------------------------------------------------

Buat struct Produk (nama, harga, persen diskon). Buat fungsi hargaPromo yang menerima struct dan mengembalikan harga setelah diskon. Tampilkan produk dengan potongan rupiah terbesar.

[CONTOH TAMPILAN]
Jumlah produk: 3
Produk 1 (nama harga diskon%): Kipas 350000 10
Produk 2 (nama harga diskon%): Kulkas 4200000 15
Produk 3 (nama harga diskon%): Setrika 280000 25
Kipas: Rp350000 -> Rp315000
Kulkas: Rp4200000 -> Rp3570000
Setrika: Rp280000 -> Rp210000
Potongan terbesar: Kulkas (hemat Rp630000)

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

struct Produk {
    string nama;
    long long harga;
    int diskonPersen;
};

long long hargaPromo(Produk pr) {
    return pr.harga - pr.harga * pr.diskonPersen / 100;
}

int main() {
    Produk tokoElektronik[20];
    int jumlahProduk;
    cout << "Jumlah produk: ";
    cin >> jumlahProduk;
    for (int i = 0; i < jumlahProduk; i++) {
        cout << "Produk " << i + 1 << " (nama harga diskon%): ";
        cin >> tokoElektronik[i].nama >> tokoElektronik[i].harga >> tokoElektronik[i].diskonPersen;
    }
    int idxHemat = 0;
    for (int i = 0; i < jumlahProduk; i++) {
        cout << tokoElektronik[i].nama << ": Rp" << tokoElektronik[i].harga << " -> Rp" << hargaPromo(tokoElektronik[i]) << endl;
        if (tokoElektronik[i].harga - hargaPromo(tokoElektronik[i]) > tokoElektronik[idxHemat].harga - hargaPromo(tokoElektronik[idxHemat])) idxHemat = i;
    }
    cout << "Potongan terbesar: " << tokoElektronik[idxHemat].nama << " (hemat Rp" << tokoElektronik[idxHemat].harga - hargaPromo(tokoElektronik[idxHemat]) << ")" << endl;
    return 0;
}

#include <iostream>
using namespace std;

struct Kartu {
    int nilai;
    char jenis;
};

int kekuatanJenis(char j) {
    if (j == 'S') return 4;
    if (j == 'H') return 3;
    if (j == 'W') return 2;
    return 1;
}

bool lebihTinggi(Kartu a, Kartu b) {
    if (a.nilai != b.nilai) return a.nilai > b.nilai;
    return kekuatanJenis(a.jenis) > kekuatanJenis(b.jenis);
}

int main() {
    Kartu mejaRemi[10];
    int kartuDiMeja;
    cout << "Jumlah kartu: ";
    cin >> kartuDiMeja;
    for (int i = 0; i < kartuDiMeja; i++) {
        cout << "Kartu " << i + 1 << " (nilai jenis): ";
        cin >> mejaRemi[i].nilai >> mejaRemi[i].jenis;
    }
    int idxRaja = 0;
    for (int i = 1; i < kartuDiMeja; i++) if (lebihTinggi(mejaRemi[i], mejaRemi[idxRaja])) idxRaja = i;
    cout << "Kartu tertinggi: kartu ke-" << idxRaja + 1 << " (" << mejaRemi[idxRaja].nilai << " " << mejaRemi[idxRaja].jenis << ")" << endl;
    return 0;
}

#include <iostream>
#include <string>
using namespace std;

struct Pahlawan {
    string nama;
    int hp;
    int hpMaks;
    int serangan;
};

bool primaRawat(int n) {
    if (n < 2) return false;
    for (int d = 2; d * d <= n; d++) if (n % d == 0) return false;
    return true;
}

bool kuadratRawat(int n) {
    for (int a = 1; a * a <= n; a++) if (a * a == n) return true;
    return false;
}

int main() {
    Pahlawan tabib = {"Tabib Kuadrat", 40, 40, 12};
    int hpLintah = 60;
    int giliranRawat = 1, ujungBawahR = 1, ujungAtasR = 10;
    while (tabib.hp > 0 && hpLintah > 0) {
        int angkaRamuan;
        cout << "\nGiliran " << giliranRawat << ":\n";
        cout << "Masukkan angka " << ujungBawahR << "-" << ujungAtasR << ": ";
        cin >> angkaRamuan;
        if (angkaRamuan < ujungBawahR || angkaRamuan > ujungAtasR) {
            cout << "Di luar rentang, ulangi!\n";
            continue;
        }
        if (primaRawat(angkaRamuan)) {
            hpLintah -= tabib.serangan;
            if (hpLintah < 0) hpLintah = 0;
            cout << "Prima! Tabib menyerang " << tabib.serangan << " damage. HP Lintah: " << hpLintah << endl;
        } else if (kuadratRawat(angkaRamuan)) {
            tabib.hp += 10;
            if (tabib.hp > tabib.hpMaks) tabib.hp = tabib.hpMaks;
            cout << "Kuadrat sempurna! Tabib menyembuhkan diri. HP Tabib: " << tabib.hp << endl;
        } else {
            int hisapan = angkaRamuan % 10 + 1;
            tabib.hp -= hisapan;
            if (tabib.hp < 0) tabib.hp = 0;
            cout << "Lintah menghisap " << hisapan << " darah. HP Tabib: " << tabib.hp << endl;
        }
        giliranRawat++;
        ujungBawahR += 10;
        ujungAtasR += 10;
    }
    if (hpLintah == 0) cout << "\nLintah Raksasa kering! Tabib Kuadrat menang!\n";
    else cout << "\nTabib Kuadrat pingsan... Lintah Raksasa menang!\n";
    return 0;
}

            int jurusPendekar = duoPendekar[idxMaju].serangan + angkaJurusSiluman % 5;
            rajaSiluman.hp -= jurusPendekar;
            if (rajaSiluman.hp < 0) rajaSiluman.hp = 0;
            cout << duoPendekar[idxMaju].nama << " menyerang " << jurusPendekar << " damage! HP Raja Siluman: " << rajaSiluman.hp << endl;
        } else {
            duoPendekar[idxMaju].hp -= rajaSiluman.serangan;
            if (duoPendekar[idxMaju].hp < 0) duoPendekar[idxMaju].hp = 0;
            cout << "Raja Siluman mencakar " << duoPendekar[idxMaju].nama << " " << rajaSiluman.serangan << " damage! HP: " << duoPendekar[idxMaju].hp << endl;
        }
        giliranSiluman++;
        sisiBawah += 10;
        sisiAtas += 10;
    }
    if (rajaSiluman.hp == 0) cout << "\nRaja Siluman musnah! Duo Pendekar menang!\n";
    else cout << "\nKedua pendekar tumbang... Raja Siluman menang!\n";
    return 0;
}

Soal 48. Pecahan Campuran Tukang Kayu
----------------------------------------------------------------------

Buat struct Pecahan dan fungsi sederhanakan yang mengembalikan struct. Tampilkan pecahan dalam bentuk paling sederhana, dan jika pembilang lebih besar dari penyebut, tampilkan juga sebagai pecahan campuran.

[CONTOH TAMPILAN]
Pecahan (pembilang penyebut): 42 8
Bentuk sederhana: 21/4
Pecahan campuran: 5 1/4

[JAWABAN]
#include <iostream>
using namespace std;

struct Pecahan {
    int atas;
    int bawah;
};

Pecahan sederhanakan(Pecahan pc) {
    int a = pc.atas, b = pc.bawah;
    while (b != 0) { int s = a % b; a = b; b = s; }
    Pecahan hasilSederhana = {pc.atas / a, pc.bawah / a};
    return hasilSederhana;
}

int main() {
    Pecahan ukuranPapan;
    cout << "Pecahan (pembilang penyebut): ";
    cin >> ukuranPapan.atas >> ukuranPapan.bawah;
    Pecahan papanRapi = sederhanakan(ukuranPapan);
    cout << "Bentuk sederhana: " << papanRapi.atas << "/" << papanRapi.bawah << endl;
    if (papanRapi.atas >= papanRapi.bawah) {
        int bulat = papanRapi.atas / papanRapi.bawah, sisaPecahan = papanRapi.atas % papanRapi.bawah;
        cout << "Pecahan campuran: " << bulat;
        if (sisaPecahan != 0) cout << " " << sisaPecahan << "/" << papanRapi.bawah;
        cout << endl;
    }
    return 0;
}

Soal 49. Kasir Toko Bangunan
----------------------------------------------------------------------

Buat struct BarangBangunan (kode, nama, harga). Isi daftar barang langsung di program. Pembeli memasukkan kode dan jumlah (kode X untuk selesai). Simpan item belanja sebagai struct ItemBelanja, lalu cetak struk lengkap dengan total.

[CONTOH TAMPILAN]
Kode barang (X selesai): SM 3
Kode barang (X selesai): PS 10
Kode barang (X selesai): XX 1
Kode XX tidak ada!
Kode barang (X selesai): CT 2
Kode barang (X selesai): X
===== STRUK TOKO BANGUNAN =====
Semen 40kg x3 = Rp186000
Pasir per karung x10 = Rp180000
Cat tembok 5kg x2 = Rp270000
TOTAL: Rp636000

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

struct BarangBangunan {
    string kode;
    string nama;
    long long harga;
};

struct ItemBelanja {
    int idxBarang;
    int jumlah;
};

int main() {
    BarangBangunan katalogBangunan[4] = {{"SM", "Semen 40kg", 62000}, {"PS", "Pasir per karung", 18000}, {"CT", "Cat tembok 5kg", 135000}, {"PK", "Paku 1kg", 25000}};
    ItemBelanja keranjangBangunan[20];
    int isiKeranjang = 0;
    string kodeBeli;
    cout << "Kode barang (X selesai): ";
    cin >> kodeBeli;
    while (kodeBeli != "X") {
        int jumlahBeli;
        cin >> jumlahBeli;
        int idxKetemu = -1;
        for (int i = 0; i < 4; i++) if (katalogBangunan[i].kode == kodeBeli) idxKetemu = i;
        if (idxKetemu == -1) cout << "Kode " << kodeBeli << " tidak ada!" << endl;
        else {
            keranjangBangunan[isiKeranjang].idxBarang = idxKetemu;
            keranjangBangunan[isiKeranjang].jumlah = jumlahBeli;
            isiKeranjang++;
        }
        cout << "Kode barang (X selesai): ";
        cin >> kodeBeli;
    }
    long long totalBangunan = 0;
    cout << "===== STRUK TOKO BANGUNAN =====" << endl;
    for (int i = 0; i < isiKeranjang; i++) {
        BarangBangunan b = katalogBangunan[keranjangBangunan[i].idxBarang];
        long long sub = b.harga * keranjangBangunan[i].jumlah;
        totalBangunan += sub;
        cout << b.nama << " x" << keranjangBangunan[i].jumlah << " = Rp" << sub << endl;
    }
    cout << "TOTAL: Rp" << totalBangunan << endl;
    return 0;
}

Soal 50. Agenda Balai Desa
----------------------------------------------------------------------

Buat struct Acara (nama acara, tanggal bertipe struct Tanggal, jam mulai). Simpan beberapa acara lalu tampilkan semua acara pada tanggal yang dicari.

[CONTOH TAMPILAN]
Jumlah acara: 4
Acara 1 (nama h b t jam): Rapat_RT 12 10 2026 19
Acara 2 (nama h b t jam): Posyandu 15 10 2026 8
Acara 3 (nama h b t jam): Senam_Lansia 12 10 2026 6
Acara 4 (nama h b t jam): Pengajian 20 10 2026 20
Cari tanggal (h b t): 12 10 2026
- Rapat_RT pukul 19.00
- Senam_Lansia pukul 6.00

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

struct Tanggal {
    int hari, bulan, tahun;
};

struct Acara {
    string nama;
    Tanggal tanggal;
    int jamMulai;
};

bool tanggalSama(Tanggal a, Tanggal b) {
    return a.hari == b.hari && a.bulan == b.bulan && a.tahun == b.tahun;
}

int main() {
    Acara agendaBalai[30];
    int jumlahAcara;
    Tanggal tanggalCari;
    cout << "Jumlah acara: ";
    cin >> jumlahAcara;
    for (int i = 0; i < jumlahAcara; i++) {
        cout << "Acara " << i + 1 << " (nama h b t jam): ";
        cin >> agendaBalai[i].nama >> agendaBalai[i].tanggal.hari >> agendaBalai[i].tanggal.bulan >> agendaBalai[i].tanggal.tahun >> agendaBalai[i].jamMulai;
    }
    cout << "Cari tanggal (h b t): ";
    cin >> tanggalCari.hari >> tanggalCari.bulan >> tanggalCari.tahun;
    int acaraKetemu = 0;
    for (int i = 0; i < jumlahAcara; i++) {
        if (tanggalSama(agendaBalai[i].tanggal, tanggalCari)) {
            cout << "- " << agendaBalai[i].nama << " pukul " << agendaBalai[i].jamMulai << ".00" << endl;
            acaraKetemu++;
        }
    }
    if (acaraKetemu == 0) cout << "Tidak ada acara di tanggal itu." << endl;
    return 0;
}
