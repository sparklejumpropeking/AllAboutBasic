Soal 1. Tukang Karpet Istana
----------------------------------------------------------------------

Buat fungsi yang mengembalikan luas persegi panjang. Gunakan untuk menghitung luas karpet dan harga karpet (Rp75.000 per m²).

[JAWABAN]

    return 0;
}

----------------------------------------------------------------------

Tiap keluarga: 2 jagung, 3 padi

using namespace std;

int fpbPanen(int panenJagung, int panenPadi) {
    if (panenPadi == 0) return panenJagung;

    cin >> karungJagung >> karungPadi;
    int keluargaPenerima = fpbPanen(karungJagung, karungPadi);

    return 0;
}

Soal 11. Jadwal Kapal Bertemu di Pelabuhan (KPK)
----------------------------------------------------------------------

Kapal A singgah tiap a hari, kapal B tiap b hari. Buat fungsi FPB dan fungsi KPK yang memanggil FPB: KPK = a / FPB × b.

[CONTOH TAMPILAN]
Siklus singgah kapal A dan B (hari): 6 15
Kedua kapal bertemu lagi setelah 30 hari

[JAWABAN]
#include <iostream>
using namespace std;

int fpbPelabuhan(int siklusA, int siklusB) {
    while (siklusB != 0) {
        int sisaSiklus = siklusA % siklusB;
        siklusA = siklusB;
        siklusB = sisaSiklus;
    }
    return siklusA;
}

int kpkPelabuhan(int siklusA, int siklusB) {
    return siklusA / fpbPelabuhan(siklusA, siklusB) * siklusB;
}

int main() {
    int jadwalKapalA, jadwalKapalB;
    cout << "Siklus singgah kapal A dan B (hari): ";
    cin >> jadwalKapalA >> jadwalKapalB;
    cout << "Kedua kapal bertemu lagi setelah " << kpkPelabuhan(jadwalKapalA, jadwalKapalB) << " hari" << endl;
    return 0;
}

using namespace std;

bool bintangPrima(int cahayaBintang) {
    if (cahayaBintang < 2) return false;
    for (int teropong = 2; teropong * teropong <= cahayaBintang; teropong++)
        if (cahayaBintang % teropong == 0) return false;
    return true;
}

void petaBintang(int langitAwal, int langitAkhir) {
    cout << "Bintang prima:";
    for (int titikLangit = langitAwal; titikLangit <= langitAkhir; titikLangit++)
        if (bintangPrima(titikLangit)) cout << " " << titikLangit;
    cout << endl;
}

int main() {
    int batasAwalLangit, batasAkhirLangit;
    cout << "Rentang langit (awal akhir): ";
    cin >> batasAwalLangit >> batasAkhirLangit;
    petaBintang(batasAwalLangit, batasAkhirLangit);
    return 0;
}

Soal 14. Prasasti Bolak-Balik
----------------------------------------------------------------------

Buat fungsi pembalik angka dan fungsi bool palindrom yang memakai fungsi pembalik tersebut.

[CONTOH TAMPILAN]
Angka di prasasti: 45654
Dibaca terbalik: 45654
Prasasti simetris (palindrom)!

[JAWABAN]
#include <iostream>
using namespace std;

long long balikPrasasti(long long ukiranAngka) {
    long long ukiranTerbalik = 0;
    while (ukiranAngka > 0) {
        ukiranTerbalik = ukiranTerbalik * 10 + ukiranAngka % 10;
        ukiranAngka /= 10;
    }
    return ukiranTerbalik;
}

bool prasastiSimetris(long long ukiranAngka) {
    return ukiranAngka == balikPrasasti(ukiranAngka);
}

int main() {
    long long angkaPrasasti;
    cout << "Angka di prasasti: ";
    cin >> angkaPrasasti;
    cout << "Dibaca terbalik: " << balikPrasasti(angkaPrasasti) << endl;
    if (prasastiSimetris(angkaPrasasti)) cout << "Prasasti simetris (palindrom)!" << endl;
    else cout << "Prasasti tidak simetris." << endl;
    return 0;
}

        if (angkaLab % pembagiLab == 0) totalFaktorLab += pembagiLab;

bool angkaArmstrong(int angkaLab) {
    int digitLab = 0, salinanLab = angkaLab, totalPangkatLab = 0;

        totalPangkatLab += pangkatDigit;
        salinanLab /= 10;
    }
    return totalPangkatLab == angkaLab;
}

        cin >> hurufDiucap;

#include <iostream>
#include <iomanip>
using namespace std;

int waktuTempuh(double jarakRel, double lajuKeretaMalam) {
    return (int)(jarakRel / lajuKeretaMalam * 60);
}

void cetakJam(int menitSejakTengahMalam) {
    menitSejakTengahMalam %= 24 * 60;
    cout << setfill('0') << setw(2) << menitSejakTengahMalam / 60 << ":" << setw(2) << menitSejakTengahMalam % 60 << endl;
}

int main() {
    int jamBerangkatKereta, menitBerangkatKereta;
