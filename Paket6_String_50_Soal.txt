Soal 5. Analisis Isi Telegram

[JAWABAN]

#include <string>
#include <cctype>
using namespace std;

int main() {
    string isiTelegram;
    int vokalTele = 0, konsonanTele = 0, angkaTele = 0, spasiTele = 0, lainTele = 0;
    cout << "Isi telegram: ";
    getline(cin, isiTelegram);
    for (int i = 0; i < (int)isiTelegram.length(); i++) {
        char hurufTele = tolower(isiTelegram[i]);
        if (isalpha(hurufTele)) {
            if (hurufTele == 'a' || hurufTele == 'i' || hurufTele == 'u' || hurufTele == 'e' || hurufTele == 'o') vokalTele++;
            else konsonanTele++;
        } else if (isdigit(hurufTele)) angkaTele++;
        else if (hurufTele == ' ') spasiTele++;
        else lainTele++;
    }
    cout << "Vokal: " << vokalTele << ", Konsonan: " << konsonanTele << ", Angka: " << angkaTele
         << ", Spasi: " << spasiTele << ", Lainnya: " << lainTele << endl;
    return 0;
}

Soal 13. Revisi Naskah Drama
----------------------------------------------------------------------

Sutradara mengganti nama tokoh di naskah. Ganti semua kemunculan kata lama dengan kata baru memakai find dan replace.

[CONTOH TAMPILAN]
Naskah: Bawang Merah memanggil Bawang Putih, lalu Bawang Putih menjawab
Kata lama: Putih
Kata baru: Emas
Hasil revisi: Bawang Merah memanggil Bawang Emas, lalu Bawang Emas menjawab
Diganti 2 kali

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

int main() {
    string naskahDrama, tokohLama, tokohBaru;
    cout << "Naskah: ";
    getline(cin, naskahDrama);
    cout << "Kata lama: ";
    getline(cin, tokohLama);
    cout << "Kata baru: ";
    getline(cin, tokohBaru);
    int jumlahGanti = 0;
    size_t posGanti = naskahDrama.find(tokohLama);
    while (posGanti != string::npos) {
        naskahDrama.replace(posGanti, tokohLama.length(), tokohBaru);
        jumlahGanti++;
        posGanti = naskahDrama.find(tokohLama, posGanti + tokohBaru.length());
    }
    cout << "Hasil revisi: " << naskahDrama << endl;
    cout << "Diganti " << jumlahGanti << " kali" << endl;
    return 0;
}

Soal 16. Kurir Rahasia Dua Arah (Caesar)
----------------------------------------------------------------------

Buat program yang bisa menyandikan (E) dan membuka sandi (D) dengan geseran k. Membuka sandi = menggeser sebanyak 26 − k.

[CONTOH TAMPILAN]
Mode (E=sandikan, D=buka): D
Kunci geser: 3
Teks: Shuwhpxdq Udkdvld
Hasil: Pertemuan Rahasia

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

int main() {
    char modeKurir;
    int geserKurir;
    string teksKurir;
    cout << "Mode (E=sandikan, D=buka): ";
    cin >> modeKurir;
    cout << "Kunci geser: ";
    cin >> geserKurir;
    cin.ignore();
    cout << "Teks: ";
    getline(cin, teksKurir);
    geserKurir %= 26;
    if (modeKurir == 'D' || modeKurir == 'd') geserKurir = 26 - geserKurir;
    for (int i = 0; i < (int)teksKurir.length(); i++) {
        if (teksKurir[i] >= 'a' && teksKurir[i] <= 'z') teksKurir[i] = (teksKurir[i] - 'a' + geserKurir) % 26 + 'a';
        else if (teksKurir[i] >= 'A' && teksKurir[i] <= 'Z') teksKurir[i] = (teksKurir[i] - 'A' + geserKurir) % 26 + 'A';
    }
    cout << "Hasil: " << teksKurir << endl;
    return 0;
}

Soal 19. Petugas Pos Pemeriksa Email
----------------------------------------------------------------------

Email dianggap sah jika: tepat satu '@', ada karakter sebelum '@', dan ada titik setelah '@' yang tidak berada di akhir. Jika sah, tampilkan nama pengguna dan domainnya.

[CONTOH TAMPILAN]
Alamat email: kurir.desa@posindo.id
Email sah.
Pengguna: kurir.desa
Domain  : posindo.id

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

int main() {
    string alamatSurel;
    cout << "Alamat email: ";
    cin >> alamatSurel;
    int jumlahAt = 0;
    for (int i = 0; i < (int)alamatSurel.length(); i++) if (alamatSurel[i] == '@') jumlahAt++;
    size_t posAt = alamatSurel.find('@');
    size_t posTitik = (posAt == string::npos) ? string::npos : alamatSurel.find('.', posAt);
    if (jumlahAt == 1 && posAt > 0 && posTitik != string::npos && posTitik != alamatSurel.length() - 1 && posTitik > posAt + 1) {
        cout << "Email sah." << endl;
        cout << "Pengguna: " << alamatSurel.substr(0, posAt) << endl;
        cout << "Domain  : " << alamatSurel.substr(posAt + 1) << endl;
    } else {
        cout << "Email tidak sah!" << endl;
    }
    return 0;
}

Soal 20. Penjaga Brankas Kata Sandi
----------------------------------------------------------------------

Kata sandi kuat jika panjang minimal 8, punya huruf besar, huruf kecil, dan angka. Tampilkan syarat apa saja yang belum terpenuhi.

[CONTOH TAMPILAN]
Kata sandi: kerajaan99
- Butuh huruf besar
Kata sandi lemah!

[JAWABAN]
#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    string sandiBrankas;
    bool adaBesar = false, adaKecil = false, adaAngka = false;
    cout << "Kata sandi: ";
    cin >> sandiBrankas;
    for (int i = 0; i < (int)sandiBrankas.length(); i++) {
        if (isupper(sandiBrankas[i])) adaBesar = true;
        else if (islower(sandiBrankas[i])) adaKecil = true;
        else if (isdigit(sandiBrankas[i])) adaAngka = true;
    }
    bool sandiKuat = true;
    if (sandiBrankas.length() < 8) { cout << "- Minimal 8 karakter" << endl; sandiKuat = false; }
    if (!adaBesar) { cout << "- Butuh huruf besar" << endl; sandiKuat = false; }
    if (!adaKecil) { cout << "- Butuh huruf kecil" << endl; sandiKuat = false; }
    if (!adaAngka) { cout << "- Butuh angka" << endl; sandiKuat = false; }
    if (sandiKuat) cout << "Kata sandi kuat, brankas terkunci aman!" << endl;
    else cout << "Kata sandi lemah!" << endl;
    return 0;
}

    for (int i = 0; i < (int)kataPertama.length(); i++) if (isalpha(kataPertama[i])) hitungHuruf[tolower(kataPertama[i]) - 'a']++;
    for (int i = 0; i < (int)kataKedua.length(); i++) if (isalpha(kataKedua[i])) hitungHuruf[tolower(kataKedua[i]) - 'a']--;
    bool anagram = true;

Soal 24. Pemadat Pesan Telegraf (Run-Length Encoding)

Telegraf mahal per huruf, jadi huruf yang berulang berturut-turut dipadatkan: "aaabbc" menjadi "a3b2c1".

[CONTOH TAMPILAN]

#include <string>
using namespace std;

    string pesanPanjang, pesanPadat = "";
    cout << "Pesan: ";
    cin >> pesanPanjang;
    int idxPesan = 0;
    while (idxPesan < (int)pesanPanjang.length()) {
        char hurufUlang = pesanPanjang[idxPesan];
        int jumlahUlang = 0;
        while (idxPesan < (int)pesanPanjang.length() && pesanPanjang[idxPesan] == hurufUlang) {
            jumlahUlang++;
            idxPesan++;
        }
        pesanPadat += hurufUlang + to_string(jumlahUlang);
    }
    cout << "Pesan padat: " << pesanPadat << endl;
    return 0;
}

Soal 43. Kata Tanpa Huruf Kembar (Isogram)
----------------------------------------------------------------------

Isogram adalah kata tanpa huruf yang berulang. Cek kata yang dimasukkan, dan jika bukan isogram tampilkan huruf pertama yang berulang.

[CONTOH TAMPILAN]
Kata: Pelangi
Pelangi adalah isogram!

[JAWABAN]
#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    string kataUjiIsogram;
    bool sudahMuncul[26] = {false};
    cout << "Kata: ";
    cin >> kataUjiIsogram;
    for (int i = 0; i < (int)kataUjiIsogram.length(); i++) {
        if (!isalpha(kataUjiIsogram[i])) continue;
        int idxHuruf = tolower(kataUjiIsogram[i]) - 'a';
        if (sudahMuncul[idxHuruf]) {
            cout << "Bukan isogram, huruf '" << char('a' + idxHuruf) << "' berulang" << endl;
            return 0;
        }
        sudahMuncul[idxHuruf] = true;
    }
    cout << kataUjiIsogram << " adalah isogram!" << endl;
    return 0;
}

Soal 44. Kalimat Lengkap Guru Mengetik (Pangram)
----------------------------------------------------------------------

Pangram adalah kalimat yang memuat semua huruf a–z. Cek kalimat dan tampilkan huruf-huruf yang belum ada.

[CONTOH TAMPILAN]
Kalimat: Kucing nakal itu lompat ke atas meja
Bukan pangram. Huruf yang belum ada: bdfhqrvwxyz

[JAWABAN]
#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    string kalimatGuru;
    bool hurufAda[26] = {false};
    cout << "Kalimat: ";
    getline(cin, kalimatGuru);
    for (int i = 0; i < (int)kalimatGuru.length(); i++)
        if (isalpha(kalimatGuru[i])) hurufAda[tolower(kalimatGuru[i]) - 'a'] = true;
    string hurufHilang = "";
    for (int h = 0; h < 26; h++) if (!hurufAda[h]) hurufHilang += char('a' + h);
    if (hurufHilang == "") cout << "Kalimat ini pangram!" << endl;
    else cout << "Bukan pangram. Huruf yang belum ada: " << hurufHilang << endl;
    return 0;
}

Soal 45. Tebak Kata Tiang Gantungan
----------------------------------------------------------------------

Kata rahasianya "pelangi". Pemain menebak satu huruf setiap giliran; huruf yang benar dibuka, salah mengurangi nyawa. Nyawa awal 5. Menang jika semua huruf terbuka.

[CONTOH TAMPILAN]
Kata: _______ | Nyawa: 5 | Tebak huruf: a
Kata: ___a___ | Nyawa: 5 | Tebak huruf: e
Kata: _e_a___ | Nyawa: 5 | Tebak huruf: z
Huruf z tidak ada!
Kata: _e_a___ | Nyawa: 4 | Tebak huruf: l
Kata: _ela___ | Nyawa: 4 | Tebak huruf: i
Kata: _ela__i | Nyawa: 4 | Tebak huruf: p
Kata: pela__i | Nyawa: 4 | Tebak huruf: n
Kata: pelan_i | Nyawa: 4 | Tebak huruf: g
Selamat! Kata rahasianya pelangi

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

int main() {
    string kataRahasiaGantung = "pelangi";
    string tampilanGantung(kataRahasiaGantung.length(), '_');
    int nyawaGantung = 5;
    while (nyawaGantung > 0 && tampilanGantung != kataRahasiaGantung) {
        char tebakanGantung;
        cout << "Kata: " << tampilanGantung << " | Nyawa: " << nyawaGantung << " | Tebak huruf: ";
        cin >> tebakanGantung;
        bool hurufKena = false;
        for (int i = 0; i < (int)kataRahasiaGantung.length(); i++) {
            if (kataRahasiaGantung[i] == tebakanGantung && tampilanGantung[i] == '_') {
                tampilanGantung[i] = tebakanGantung;
                hurufKena = true;
            }
        }
        if (!hurufKena) {
            nyawaGantung--;
            cout << "Huruf " << tebakanGantung << " tidak ada!" << endl;
        }
    }
    if (tampilanGantung == kataRahasiaGantung) cout << "Selamat! Kata rahasianya " << kataRahasiaGantung << endl;
    else cout << "Nyawa habis! Kata rahasianya " << kataRahasiaGantung << endl;
    return 0;
}

Soal 46. Penyair Vokal vs Monster Konsonan (pola game Quest IV)
----------------------------------------------------------------------

Penyair Vokal (HP 40) melawan Monster Konsonan (HP 50). Setiap giliran user memasukkan satu kata yang panjangnya harus di rentang tertentu: mulai 3–5 huruf, lalu rentang naik 2 tiap giliran. Jika jumlah vokal dalam kata ≥ jumlah konsonan, penyair menyerang dengan damage = vokal × 5. Jika tidak, monster menyerang dengan damage = jumlah konsonan.

[CONTOH TAMPILAN]
Giliran 1:
Masukkan kata 3-5 huruf: aku
Vokal: 2, Konsonan: 1
Penyair melantunkan syair dengan 10 damage!
HP Monster Konsonan: 40

Giliran 2:
Masukkan kata 5-7 huruf: sastra
Vokal: 2, Konsonan: 4
Monster Konsonan menggeram dengan 4 damage!
HP Penyair Vokal: 36

Giliran 3:
Masukkan kata 7-9 huruf: kelapa
Panjang kata tidak sesuai, ulangi!

Giliran 3:
Masukkan kata 7-9 huruf: keluarga
Vokal: 4, Konsonan: 4
Penyair melantunkan syair dengan 20 damage!
HP Monster Konsonan: 20

Giliran 4:
Masukkan kata 9-11 huruf: keadaannya
Vokal: 5, Konsonan: 5
Penyair melantunkan syair dengan 25 damage!
HP Monster Konsonan: 0

Monster Konsonan bungkam! Penyair Vokal menang!

[JAWABAN]
#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    int hpPenyair = 40, hpMonsterKonsonan = 50;
    int giliranSyair = 1, panjangMin = 3, panjangMaks = 5;
    while (hpPenyair > 0 && hpMonsterKonsonan > 0) {
        string kataSyair;
        cout << "\nGiliran " << giliranSyair << ":\n";
        cout << "Masukkan kata " << panjangMin << "-" << panjangMaks << " huruf: ";
        cin >> kataSyair;
        if ((int)kataSyair.length() < panjangMin || (int)kataSyair.length() > panjangMaks) {
            cout << "Panjang kata tidak sesuai, ulangi!\n";
            continue;
        }
        int vokalSyair = 0, konsonanSyair = 0;
        for (int i = 0; i < (int)kataSyair.length(); i++) {
            char hurufSyair = tolower(kataSyair[i]);
            if (!isalpha(hurufSyair)) continue;
            if (hurufSyair == 'a' || hurufSyair == 'i' || hurufSyair == 'u' || hurufSyair == 'e' || hurufSyair == 'o') vokalSyair++;
            else konsonanSyair++;
        }
        cout << "Vokal: " << vokalSyair << ", Konsonan: " << konsonanSyair << endl;
        if (vokalSyair >= konsonanSyair) {
            int bait = vokalSyair * 5;
            hpMonsterKonsonan -= bait;
            if (hpMonsterKonsonan < 0) hpMonsterKonsonan = 0;
            cout << "Penyair melantunkan syair dengan " << bait << " damage!\n";
            cout << "HP Monster Konsonan: " << hpMonsterKonsonan << endl;
        } else {
            hpPenyair -= konsonanSyair;
            if (hpPenyair < 0) hpPenyair = 0;
            cout << "Monster Konsonan menggeram dengan " << konsonanSyair << " damage!\n";
            cout << "HP Penyair Vokal: " << hpPenyair << endl;
        }
        giliranSyair++;
        panjangMin += 2;
        panjangMaks += 2;
    }
    if (hpMonsterKonsonan == 0) cout << "\nMonster Konsonan bungkam! Penyair Vokal menang!\n";
    else cout << "\nPenyair Vokal kehabisan kata... Monster Konsonan menang!\n";
    return 0;
}

Soal 47. Ksatria Palindrom vs Penyihir Acak (pola game Quest IV)
----------------------------------------------------------------------

Ksatria Palindrom (HP 30) melawan Penyihir Acak (HP 45). Setiap giliran user memasukkan kata dengan panjang minimal tertentu: mulai minimal 3 huruf dan bertambah 1 setiap giliran. Jika kata palindrom (abaikan besar kecil), ksatria menyerang dengan damage = panjang kata × 3. Jika bukan, penyihir menyerang dengan damage = panjang kata.

[CONTOH TAMPILAN]
Giliran 1:
Masukkan kata minimal 3 huruf: ada
ada palindrom! Ksatria menebas 9 damage!
HP Penyihir Acak: 36

Giliran 2:
Masukkan kata minimal 4 huruf: kodok
kodok palindrom! Ksatria menebas 15 damage!
HP Penyihir Acak: 21

Giliran 3:
Masukkan kata minimal 5 huruf: makan
makan bukan palindrom.. Penyihir mengutuk 5 damage!
HP Ksatria Palindrom: 25

Giliran 4:
Masukkan kata minimal 6 huruf: katak
Kata terlalu pendek, ulangi!

Giliran 4:
Masukkan kata minimal 6 huruf: kasurrusak
kasurrusak palindrom! Ksatria menebas 30 damage!
HP Penyihir Acak: 0

Penyihir Acak lenyap! Ksatria Palindrom menang!

[JAWABAN]
#include <iostream>
#include <string>
#include <cctype>
using namespace std;

int main() {
    int hpKsatriaPalindrom = 30, hpPenyihirAcak = 45;
    int giliranKata = 1, minimalHuruf = 3;
    while (hpKsatriaPalindrom > 0 && hpPenyihirAcak > 0) {
        string kataPedang;
        cout << "\nGiliran " << giliranKata << ":\n";
        cout << "Masukkan kata minimal " << minimalHuruf << " huruf: ";
        cin >> kataPedang;
        if ((int)kataPedang.length() < minimalHuruf) {
            cout << "Kata terlalu pendek, ulangi!\n";
            continue;
        }
        bool kataCermin = true;
        for (int i = 0, j = kataPedang.length() - 1; i < j; i++, j--)
            if (tolower(kataPedang[i]) != tolower(kataPedang[j])) { kataCermin = false; break; }
        int panjangPedang = kataPedang.length();
        if (kataCermin) {
            hpPenyihirAcak -= panjangPedang * 3;
            if (hpPenyihirAcak < 0) hpPenyihirAcak = 0;
            cout << kataPedang << " palindrom! Ksatria menebas " << panjangPedang * 3 << " damage!\n";
            cout << "HP Penyihir Acak: " << hpPenyihirAcak << endl;
        } else {
            hpKsatriaPalindrom -= panjangPedang;
            if (hpKsatriaPalindrom < 0) hpKsatriaPalindrom = 0;
            cout << kataPedang << " bukan palindrom.. Penyihir mengutuk " << panjangPedang << " damage!\n";
            cout << "HP Ksatria Palindrom: " << hpKsatriaPalindrom << endl;
        }
        giliranKata++;
        minimalHuruf++;
    }
    if (hpPenyihirAcak == 0) cout << "\nPenyihir Acak lenyap! Ksatria Palindrom menang!\n";
    else cout << "\nKsatria Palindrom kalah... Penyihir Acak menang!\n";
    return 0;
}

Soal 48. Arsiparis Pemilah Berkas
----------------------------------------------------------------------

Arsiparis memilah berkas berdasarkan ekstensi nama file (bagian setelah titik terakhir): docx/pdf/txt = dokumen, jpg/png = gambar, cpp/py = kode program, selain itu tidak dikenal.

[CONTOH TAMPILAN]
Nama berkas: laporan.final.revisi.pdf
Ekstensi: pdf
Masuk lemari DOKUMEN

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

int main() {
    string namaBerkas;
    cout << "Nama berkas: ";
    cin >> namaBerkas;
    size_t titikTerakhir = namaBerkas.find_last_of('.');
    if (titikTerakhir == string::npos || titikTerakhir == namaBerkas.length() - 1) {
        cout << "Berkas tanpa ekstensi!" << endl;
        return 0;
    }
    string ekstensiBerkas = namaBerkas.substr(titikTerakhir + 1);
    cout << "Ekstensi: " << ekstensiBerkas << endl;
    if (ekstensiBerkas == "docx" || ekstensiBerkas == "pdf" || ekstensiBerkas == "txt") cout << "Masuk lemari DOKUMEN" << endl;
    else if (ekstensiBerkas == "jpg" || ekstensiBerkas == "png") cout << "Masuk lemari GAMBAR" << endl;
    else if (ekstensiBerkas == "cpp" || ekstensiBerkas == "py") cout << "Masuk lemari KODE PROGRAM" << endl;
    else cout << "Jenis berkas tidak dikenal" << endl;
    return 0;
}

Soal 49. Juri Lomba Mengarang
----------------------------------------------------------------------

Hitung jumlah kalimat dalam karangan (diakhiri ., ?, atau !) dan rata-rata jumlah kata per kalimat.

[CONTOH TAMPILAN]
Karangan: Aku suka hujan. Apakah kamu juga? Ayo bermain di luar!
Jumlah kalimat: 3
Jumlah kata: 10
Rata-rata kata per kalimat: 3.33333

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

int main() {
    string karanganMurid;
    int jumlahKalimatKarangan = 0, jumlahKataKarangan = 0;
    cout << "Karangan: ";
    getline(cin, karanganMurid);
    for (int i = 0; i < (int)karanganMurid.length(); i++) {
        char c = karanganMurid[i];
        if (c == '.' || c == '?' || c == '!') jumlahKalimatKarangan++;
        if (c != ' ' && (i == 0 || karanganMurid[i - 1] == ' ')) jumlahKataKarangan++;
    }
    cout << "Jumlah kalimat: " << jumlahKalimatKarangan << endl;
    cout << "Jumlah kata: " << jumlahKataKarangan << endl;
    if (jumlahKalimatKarangan > 0) cout << "Rata-rata kata per kalimat: " << (double)jumlahKataKarangan / jumlahKalimatKarangan << endl;
    return 0;
}

Soal 50. Durasi Shift Satpam
----------------------------------------------------------------------

Jam masuk dan keluar shift ditulis sebagai teks "HH:MM". Pecah dengan substr dan stoi, lalu hitung durasi shift (shift malam bisa melewati tengah malam).

[CONTOH TAMPILAN]
Jam masuk (HH:MM): 22:15
Jam pulang (HH:MM): 06:45
Durasi shift: 8 jam 30 menit

[JAWABAN]
#include <iostream>
#include <string>
using namespace std;

int main() {
    string jamMasukSatpam, jamPulangSatpam;
    cout << "Jam masuk (HH:MM): ";
    cin >> jamMasukSatpam;
    cout << "Jam pulang (HH:MM): ";
    cin >> jamPulangSatpam;
    int menitMasuk = stoi(jamMasukSatpam.substr(0, 2)) * 60 + stoi(jamMasukSatpam.substr(3, 2));
    int menitPulang = stoi(jamPulangSatpam.substr(0, 2)) * 60 + stoi(jamPulangSatpam.substr(3, 2));
    int durasiShift = menitPulang - menitMasuk;
    if (durasiShift < 0) durasiShift += 24 * 60;
    cout << "Durasi shift: " << durasiShift / 60 << " jam " << durasiShift % 60 << " menit" << endl;
    return 0;
}
